def asset_existe(path):
    try:
        ee.data.getAsset(path)
        return True
    except ee.EEException:
        return False

def propriedades_asset(path):
    try:
        return ee.data.getAsset(path).get('properties', {}) or {}
    except ee.EEException:
        return None

def info_de_propriedades(p):
    if p is None:
        return None
    rodada = p.get('rodada_gfs_ms', 0)
    return {
        'versao': p.get('versao', 'definitiva'),
        'alg': p.get('versao_algoritmo', 'V1.0'),
        'rodada': int(float(rodada)) if rodada is not None else 0,
    }

def garantir_colecao(path):

        ee.data.createAsset({'type': 'IMAGE_COLLECTION'}, path)

def listar_nomes(parent):
    nomes, token = [], None
    while True:
        params = {'parent': parent, 'pageSize': 1000}
        if token:
            params['pageToken'] = token
        resp = ee.data.listAssets(params)
        for a in resp.get('assets', []):
            caminho = a.get('id') or a.get('name')
            nomes.append(caminho.split('/')[-1])
        token = resp.get('nextPageToken')
        if not token:
            break
    return nomes

def listar_tarefas_ativas():
    """Retorna {descricao: id} de TODAS as tarefas ativas (a fila de 3000 é por usuário)."""
    ativas = {}
    for t in ee.data.getTaskList():
        if t.get('state') in ESTADOS_ATIVOS:
            ativas[t.get('description', '')] = t.get('id')
    return ativas

def promover_tmp(nome):
    """Substitui o asset definitivo pelo temporário já concluído."""
    tmp = f'{ASSET_TEMP}/{nome}'
    final = f'{ASSET_DESTINO}/{nome}'
    if not asset_existe(tmp):
        return False
    if asset_existe(final):
        ee.data.deleteAsset(final)
    ee.data.renameAsset(tmp, final)
    return True

def finalizar_trocas_pendentes(tarefas_ativas):
    if SIMULAR or not asset_existe(ASSET_TEMP):
        return
    promovidos = 0
    for nome in listar_nomes(ASSET_TEMP):
        if desc_de_nome(nome) in tarefas_ativas:
            continue
        if promover_tmp(nome):
            promovidos += 1
    if promovidos:
        print(f"🔁 {promovidos} dia(s) promovido(s) da coleção temporária para {ASSET_DESTINO}.")

def mapear_colecao():
    if not asset_existe(ASSET_DESTINO):
        return {}
    col = ee.ImageCollection(ASSET_DESTINO)

    def para_feature(img):
        d = img.toDictionary()
        return ee.Feature(None, {
            'id': img.get('system:index'),
            'versao': d.get('versao', 'definitiva'),
            'alg': d.get('versao_algoritmo', 'V1.0'),
            'rodada': d.get('rodada_gfs_ms', 0),
        })

    try:
        lista = col.map(para_feature) \
            .reduceColumns(ee.Reducer.toList().repeat(4), ['id', 'versao', 'alg', 'rodada']) \
            .get('list').getInfo()
    except Exception as e:
        raise RuntimeError(f"Falha ao inventariar {ASSET_DESTINO}; execução interrompida para evitar reenvio em massa. Detalhe: {e}")

    inventario = {}
    for idx, versao, alg, rodada in zip(*lista):
        inventario[str(idx).split('/')[-1]] = {
            'versao': versao, 'alg': alg,
            'rodada': int(float(rodada)) if rodada is not None else 0,
        }
    return inventario

def expandir_borda(img):
    return img.unmask(img.reproject(crs=crsNativo, crsTransform=transformNativo)
                      .focalMean(1, 'square', 'pixels'))

def calib_aditiva(img, m_alvo, m_modelo):
    return img.add(expandir_borda(m_alvo).subtract(expandir_borda(m_modelo)))

def fator_v10(m_alvo, m_modelo):
    """Fator exatamente como na V1.0 (sem teto)."""
    return expandir_borda(m_alvo).divide(expandir_borda(m_modelo).max(1e-3))

def calib_multiplicativa(img, m_alvo, m_modelo):
    # Onde o fator da V1.0 já era <= FATOR_MAX, min() devolve o mesmo valor: resultado idêntico
    return img.multiply(fator_v10(m_alvo, m_modelo).min(FATOR_MAX))

def teto_umidade(solo, camada):
    m = expandir_borda(solo.select(f'{camada}_m_smap'))
    s = expandir_borda(solo.select(f'{camada}_s_smap'))
    return m.add(s.multiply(SIGMAS_TETO_UMIDADE)).min(UMIDADE_MAX_FISICA).unmask(UMIDADE_MAX_FISICA)

def es_de(T):
    return T.expression('0.6108 * exp(17.625 * T / (T + 243.04))', {'T': T})

def vpd_e_ea(tmax, tmin, rh):
    es = es_de(tmax).add(es_de(tmin)).divide(2)
    ea = es.multiply(rh).divide(100)
    return es.subtract(ea).max(0).rename('vpd'), ea

def eto_penman_monteith(tmax, tmin, rh, u2, rs, data_str):
    """ETo diária FAO-56 Penman-Monteith (mm/dia), a mesma conta que a rota GFS faz desde a V1.0.
    Entradas por pixel: Tmax e Tmin (°C), UR (%), vento a 2 m (m/s) e radiação solar (MJ/m²/dia).
    Usada pelas rotas GFS e provisória (ERA5); nos dias definitivos a ETo vem do BR-DWGD."""
    data_atual = ee.Date(data_str)
    lat_rad = ee.Image.pixelLonLat().select('latitude').multiply(math.pi / 180.0)
    pressao_atm = elevacao_br.multiply(-0.0065).add(293.0).divide(293.0).pow(5.26).multiply(101.3)
    gamma_dinamico = pressao_atm.multiply(0.000665)
    tmedia = tmax.add(tmin).divide(2)
    vpd, ea = vpd_e_ea(tmax, tmin, rh)
    J = ee.Number(data_atual.getRelative('day', 'year')).add(1)
    dr = J.multiply(2.0 * math.pi / 365.0).cos().multiply(0.033).add(1.0)
    delta_sol = J.multiply(2.0 * math.pi / 365.0).subtract(1.39).sin().multiply(0.409)
    ws = lat_rad.tan().multiply(delta_sol.tan()).multiply(-1.0).clamp(-1.0, 1.0).acos()
    termo1 = ws.multiply(lat_rad.sin()).multiply(delta_sol.sin())
    termo2 = lat_rad.cos().multiply(delta_sol.cos()).multiply(ws.sin())
    Ra = termo1.add(termo2).multiply(dr).multiply((24.0 * 60.0 / math.pi) * 0.0820)
    Rso = elevacao_br.multiply(0.00002).add(0.75).multiply(Ra)
    Rs_Rso = rs.divide(Rso).clamp(0.0, 1.0)
    Rns = rs.multiply(0.77)
    tmax_k = tmax.add(273.16)
    tmin_k = tmin.add(273.16)
    rad_term = tmax_k.pow(4).add(tmin_k.pow(4)).divide(2.0).multiply(4.903e-9)
    umid_term = ea.sqrt().multiply(-0.14).add(0.34)
    nuvem_term = Rs_Rso.multiply(1.35).subtract(0.35)
    Rn = Rns.subtract(rad_term.multiply(umid_term).multiply(nuvem_term).max(0))
    delta_press = tmedia.expression(
        '4283.58 * (0.6108 * exp(17.625 * T / (T + 243.04))) / ((T + 243.04) ** 2)', {'T': tmedia})
    num_1 = delta_press.multiply(Rn).multiply(0.408)
    num_2 = gamma_dinamico.multiply(ee.Image(900).divide(tmedia.add(273))).multiply(u2).multiply(vpd)
    den = delta_press.add(gamma_dinamico.multiply(ee.Image(1).add(u2.multiply(0.34))))
    return num_1.add(num_2).divide(den).max(0).rename('et_potencial')

def finalizar(img, propriedades):
    et_pot = img.select('et_potencial')
    et_real = img.select('et_real')
    transp = img.select('transpiracao_mm')
    et_real_l = et_real.min(et_pot.unmask(et_real)).rename('et_real')
    transp_l = transp.min(et_real_l).rename('transpiracao_mm')
    saida = (img.addBands(ee.Image.cat([et_real_l, transp_l]), overwrite=True)
             .select(ORDEM_BANDAS).toFloat().clip(poligonoBrasil))
    return saida.set(propriedades)

def _componentes_era5(data_str):
    dStart = ee.Date(data_str)
    dEnd = dStart.advance(1, 'day')
    mes = data_str[5:7]
    clima = ee.Image(f'{ASSET_CLIMATOLOGIA}/Matriz_Clima_{mes}')
    solo = ee.Image(f'{ASSET_SOLO}/Matriz_Solo_{mes}')
    era = ee.Image(colecao_era5.filterDate(dStart, dEnd).first())

    def mc(p):
        return clima.select(f'{p}_m_br'), clima.select(f'{p}_m_era')

    def ms(p):
        return solo.select(f'{p}_m_smap'), solo.select(f'{p}_m_era5')

    pr_raw = era.select('total_precipitation_sum').multiply(1000).max(0)
    tmax_raw = era.select('temperature_2m_max').subtract(273.15).resample('bilinear')
    tmin_raw = era.select('temperature_2m_min').subtract(273.15).resample('bilinear')
    # Limitação conhecida: o DAILY_AGGR só traz componentes médias; a velocidade derivada
    # delas subestima o vento em dias de direção variável. Corrigir exige ERA5_LAND/HOURLY.
    u2_raw = era.select('u_component_of_wind_10m').hypot(era.select('v_component_of_wind_10m')) \
        .multiply(0.748).resample('bilinear')
    rs_raw = era.select('surface_solar_radiation_downwards_sum').divide(1e6).resample('bilinear')
    t2m = era.select('temperature_2m').subtract(273.15).resample('bilinear')
    tdew = era.select('dewpoint_temperature_2m').subtract(273.15).resample('bilinear')
    rh_raw = t2m.expression(
        '100 * (exp((17.625 * Td) / (243.04 + Td)) / exp((17.625 * T) / (243.04 + T)))',
        {'T': t2m, 'Td': tdew})
    # V1.3: a evaporação potencial do ECMWF (potential_evaporation_sum) deixou de ser usada;
    # a ETo é calculada por Penman-Monteith logo abaixo, com a meteorologia já calibrada.

    c = {}
    c['pr_mm'] = calib_multiplicativa(pr_raw, *mc('pr')).max(0).rename('pr_mm')
    c['u2'] = calib_multiplicativa(u2_raw, *mc('u2')).max(0).rename('u2')
    c['rs'] = calib_multiplicativa(rs_raw, *mc('Rs')).max(0).rename('rs')
    c['tmax_C'] = calib_aditiva(tmax_raw, *mc('Tmax')).rename('tmax_C')
    c['tmin_C'] = calib_aditiva(tmin_raw, *mc('Tmin')).rename('tmin_C')
    c['RH'] = calib_aditiva(rh_raw, *mc('RH')).clamp(0, 100).rename('RH')
    c['et_potencial'] = eto_penman_monteith(c['tmax_C'], c['tmin_C'], c['RH'], c['u2'], c['rs'], data_str)
    c['vpd'], _ = vpd_e_ea(c['tmax_C'], c['tmin_C'], c['RH'])

    c['et_real'] = era.select('total_evaporation_sum').multiply(-1000).max(0) \
        .rename('et_real').resample('bilinear')
    # ATENÇÃO: problema conhecido do ERA5-Land (ver catálogo GEE). As bandas estão trocadas:
    # evaporation_from_bare_soil_sum CONTÉM os valores de transpiração da vegetação.
    # evaporation_from_vegetation_transpiration_sum contém evaporação de águas abertas. Não trocar.
    c['transpiracao_mm'] = era.select('evaporation_from_bare_soil_sum').multiply(-1000).max(0) \
        .rename('transpiracao_mm').resample('bilinear')
    c['tsolo_C'] = era.select('soil_temperature_level_1').subtract(273.15).rename('tsolo_C').resample('bilinear')
    c['tfolha_C'] = era.select('skin_temperature').subtract(273.15).rename('tfolha_C').resample('bilinear')

    ssur_era = era.select('volumetric_soil_water_layer_1')
    swat_era = era.expression('(l1*7 + l2*21 + l3*72) / 100', {
        'l1': era.select('volumetric_soil_water_layer_1'),
        'l2': era.select('volumetric_soil_water_layer_2'),
        'l3': era.select('volumetric_soil_water_layer_3'),
    })
    rsup_era = era.select('surface_runoff_sum').multiply(1000).max(0)
    rsub_era = era.select('sub_surface_runoff_sum').multiply(1000).max(0)

    c['soil_moisture_0_5cm_m3_m3'] = calib_multiplicativa(ssur_era, *ms('ssur')).max(0) \
        .min(teto_umidade(solo, 'ssur')).rename('soil_moisture_0_5cm_m3_m3')
    c['soil_moisture_0_100cm_m3_m3'] = calib_multiplicativa(swat_era, *ms('swat')).max(0) \
        .min(teto_umidade(solo, 'swat')).rename('soil_moisture_0_100cm_m3_m3')
    c['runoff_sup_mm'] = calib_multiplicativa(rsup_era, *ms('rsup')).max(0).rename('runoff_sup_mm')
    c['runoff_sub_mm'] = calib_multiplicativa(rsub_era, *ms('rsub')).max(0).rename('runoff_sub_mm')
    return c

def processarDia_Definitivo(data_str):
    c = _componentes_era5(data_str)
    dStart = ee.Date(data_str)
    dEnd = dStart.advance(1, 'day')

    dummyBR = ee.Image.constant([-99999] * 7).rename(['pr', 'Tmax', 'Tmin', 'RH', 'u2', 'Rs', 'ET'])
    imgBR = ee.Image(colChuvaGeral.filterDate(dStart, dEnd).merge(ee.ImageCollection([dummyBR])).first())
    has_real_br = imgBR.select('pr').gt(-90000)

    tmax_br = imgBR.select('Tmax').multiply(0.00106815).add(15.0)
    tmin_br = imgBR.select('Tmin').multiply(0.00106815).add(15.0)
    rh_br = imgBR.select('RH').multiply(0.39370079).add(-0.393701).clamp(0, 100)
    vpd_br, _ = vpd_e_ea(tmax_br, tmin_br, rh_br)

    br = {
        'pr_mm': imgBR.select('pr').multiply(0.00686666).add(225.0).max(0),
        'tmax_C': tmax_br,
        'tmin_C': tmin_br,
        'RH': rh_br,
        'vpd': vpd_br,
        'et_potencial': imgBR.select('ET').multiply(0.05118110).add(-0.0511811),
        'rs': imgBR.select('Rs').multiply(0.15708661).add(-0.057087),
        'u2': imgBR.select('u2').multiply(0.05905512).add(-0.059055),
    }

    interior = ee.Image.cat([
        c[b].where(has_real_br, br[b]).rename(b) if b in br else c[b]
        for b in ORDEM_BANDAS
    ]).toFloat()
    fallback = expandir_borda(ee.Image.cat([c[b] for b in ORDEM_BANDAS]).toFloat())
    img = interior.unmask(fallback)

    return finalizar(img, {
        'system:time_start': dStart.millis(),
        'versao': 'definitiva',
        'versao_algoritmo': VERSAO_ALGORITMO,
    })

def aplicarPatch_Solo(data_str):
    """Corrige SÓ umidade do solo e escoamento de um dia gravado pela V1.0.
    As outras 12 bandas e todas as propriedades são copiadas do asset existente.
    Valor V1.0 = bruto x fator_v10, então o valor com teto = valor V1.0 x (FATOR_MAX / fator_v10)
    nos pixels onde fator_v10 > FATOR_MAX; nos demais, o valor não muda."""
    solo = ee.Image(f'{ASSET_SOLO}/Matriz_Solo_{data_str[5:7]}')
    antigo = ee.Image(f'{ASSET_DESTINO}/{id_dia(data_str)}')

    def correcao(p):
        f = fator_v10(solo.select(f'{p}_m_smap'), solo.select(f'{p}_m_era5'))
        acima = f.gt(FATOR_MAX).unmask(0)
        return ee.Image.constant(1).where(acima, ee.Image.constant(FATOR_MAX).divide(f))

    novas = []
    for banda, p in [('runoff_sup_mm', 'rsup'), ('runoff_sub_mm', 'rsub'),
                     ('soil_moisture_0_5cm_m3_m3', 'ssur'), ('soil_moisture_0_100cm_m3_m3', 'swat')]:
        b = antigo.select(banda).multiply(correcao(p))
        if p in ('ssur', 'swat'):
            b = b.min(teto_umidade(solo, p))
        novas.append(b.toFloat().rename(banda))

    return (antigo.addBands(ee.Image.cat(novas), overwrite=True)
            .select(ORDEM_BANDAS)
            .set('versao_algoritmo', VERSAO_PATCH))

_cache_bandas_gfs = {}

def bandas_matriz_gfs(mes):
    if mes not in _cache_bandas_gfs:
        _cache_bandas_gfs[mes] = ee.Image(f'{ASSET_DESTINO_MATRIZ_GFS}/Matriz_GFS_{mes}').bandNames().getInfo()

def colecao_gfs_rodada(creation_ms):
    # Pressupõe system:time_start == creation_time no NOAA/GFS0P25 (filtro indexado, rápido).
    return (ee.ImageCollection(COL_GFS)
            .filterDate(creation_ms, creation_ms + 3600 * 1000)
            .filter(ee.Filter.eq('creation_time', creation_ms)))

def contagem_esperada(dia_ms, creation_ms):
    """Horas de previsão esperadas na janela (início, fim] do dia, e blocos de chuva de 6 h."""
    h0 = int(round((dia_ms - creation_ms) / 3600000.0))
    horas = [h for h in range(h0 + 1, h0 + 25) if 0 < h <= 384 and (h <= 120 or h % 3 == 0)]
    horas_chuva = [float(h) for h in horas if h % 6 == 0]
    return len(horas), horas_chuva

def processarDia_GFS(data_str, estado_anterior, ref, creation_ms, n_img_esperado, horas_chuva):
    data_atual = ee.Date(data_str)
    t0 = data_atual.millis()
    t1 = data_atual.advance(1, 'day').millis()
    mes = data_str[5:7]

    # CORRIGIDO: janela (00Z, 24Z]. Acumulados e extremos do GFS referem-se ao período
    # que TERMINA no forecast_time.
    gfs_dia = (colecao_gfs_rodada(creation_ms)
               .filter(ee.Filter.gt('forecast_time', t0))
               .filter(ee.Filter.lte('forecast_time', t1))
               .filter(ee.Filter.gt('forecast_hours', 0)))
    chuva_blocos = gfs_dia.filter(ee.Filter.inList('forecast_hours', horas_chuva))

    qtd_img = gfs_dia.size()
    qtd_chuva = chuva_blocos.size()

    if len(horas_chuva) >= 4:
        completo = qtd_img.gte(n_img_esperado).And(qtd_chuva.gte(4))
        versao_ok = ee.Algorithms.If(completo, 'provisoria_gfs', 'provisoria_gfs_incompleta')
    else:
        versao_ok = 'provisoria_gfs_incompleta'   # dia além do horizonte da rodada

    nomes = ['total_precipitation_surface', 'maximum_temperature_2m_above_ground',
             'minimum_temperature_2m_above_ground', 'downward_shortwave_radiation_flux',
             'relative_humidity_2m_above_ground', 'wind_speed_10m_above_ground']

    def vel_vento(img):
        return img.select('u_component_of_wind_10m_above_ground') \
            .hypot(img.select('v_component_of_wind_10m_above_ground')) \
            .rename('wind_speed_10m_above_ground')

    img_chuva = ee.Image(ee.Algorithms.If(
        qtd_chuva.gt(0),
        chuva_blocos.select('total_precipitation_surface').sum(),
        ee.Image.constant(0).rename('total_precipitation_surface')))

    # Nota: radiação e UR por média simples das saídas; se o fluxo de onda curta do GFS for
    # média por bucket (como a chuva), há um viés pequeno a verificar.
    img_dia = ee.Image(ee.Algorithms.If(
        qtd_img.gt(0),
        ee.Image.cat([
            img_chuva,
            # Mesma variável usada na matriz SICSB_MATRIZ_GFS (temperatura horária instantânea)
            gfs_dia.select('temperature_2m_above_ground').max().rename('maximum_temperature_2m_above_ground'),
            gfs_dia.select('temperature_2m_above_ground').min().rename('minimum_temperature_2m_above_ground'),
            gfs_dia.select(['downward_shortwave_radiation_flux', 'relative_humidity_2m_above_ground']).mean(),
            gfs_dia.map(vel_vento).mean(),
        ]),
        ee.Image.constant([0] * 6).rename(nomes)))

    pr_raw = img_dia.select('total_precipitation_surface').max(0)
    tmax_raw = img_dia.select('maximum_temperature_2m_above_ground')
    tmin_raw = img_dia.select('minimum_temperature_2m_above_ground')
    rs_raw = img_dia.select('downward_shortwave_radiation_flux').multiply(0.0864).max(0)
    rh_raw = img_dia.select('relative_humidity_2m_above_ground').clamp(0, 100)
    u2_raw = img_dia.select('wind_speed_10m_above_ground').multiply(0.748)

    cache_gfs = ee.Image(f'{ASSET_DESTINO_MATRIZ_GFS}/Matriz_GFS_{mes}')
    bandas_cache = bandas_matriz_gfs(mes)

    def medias(banda):
        alvo = f'{banda}_m_sicsb' if f'{banda}_m_sicsb' in bandas_cache else f'{banda}_m_sab'
        return cache_gfs.select(alvo), cache_gfs.select(f'{banda}_m_gfs')

    pr = calib_multiplicativa(pr_raw, *medias('pr')).max(0).rename('pr_mm')
    tmax_c = calib_aditiva(tmax_raw, *medias('Tmax'))
    tmin_c = calib_aditiva(tmin_raw, *medias('Tmin'))
    # Amplitude mínima: preserva a média e só altera pixels com amplitude < AMPLITUDE_MIN_C
    t_med = tmax_c.add(tmin_c).divide(2)
    meia_amp = tmax_c.subtract(tmin_c).max(AMPLITUDE_MIN_C).divide(2)
    tmax = t_med.add(meia_amp).rename('tmax_C')
    tmin = t_med.subtract(meia_amp).rename('tmin_C')
    rh = calib_aditiva(rh_raw, *medias('RH')).clamp(0, 100).rename('RH')
    u2 = calib_multiplicativa(u2_raw, *medias('u2')).max(0).rename('u2')
    rs = calib_multiplicativa(rs_raw, *medias('Rs')).max(0).rename('rs')

    vpd, _ = vpd_e_ea(tmax, tmin, rh)
    # Penman-Monteith FAO-56: a mesma função usada pela rota provisória (ERA5) a partir da V1.3
    et_potencial = eto_penman_monteith(tmax, tmin, rh, u2, rs, data_str)

    # Balde de solo ancorado no último período ERA5 de cada pixel (ref):
    #  - capacidade e início da drenagem nunca abaixo da umidade do último dia ERA5
    #  - ET real = ETp x razão ET real/ETp do ERA5 recente x variação relativa da umidade
    #  - transpiração = mesma fração da ET real observada no ERA5 recente
    k_et = ref.select('k_et')
    f_transp = ref.select('f_transp')
    sw_ref = ref.select('sw_ref_100')
    cap100 = sw_ref.max(SW100_MAX_MM)
    cc100 = sw_ref.max(SW100_CC_MM).min(cap100)
    cap5 = ref.select('sw_ref_5').max(SW5_MAX_MM)

    sw100 = estado_anterior.select('sw_100').min(cap100)
    sw5 = estado_anterior.select('sw_5').min(cap5)
    saturacao = sw100.divide(cap100)
    cn = ee.Image.constant(60).add(saturacao.multiply(35))
    s_ret = ee.Image.constant(25400).divide(cn).subtract(254)
    p_ia = pr.subtract(s_ret.multiply(0.2)).max(0)
    runoff_sup = p_ia.pow(2).divide(p_ia.add(s_ret)).rename('runoff_sup_mm')
    infiltracao = pr.subtract(runoff_sup).max(0)

    g_umidade = sw100.divide(sw_ref.max(1.0)).clamp(0.5, 1.5)
    et_real = et_potencial.multiply(k_et).multiply(g_umidade).min(et_potencial).rename('et_real')
    transp_b = et_real.multiply(f_transp).rename('transpiracao_mm')
    evaporacao = et_real.subtract(transp_b)

    sw100_tmp = sw100.add(infiltracao).subtract(et_real)
    excesso = sw100_tmp.subtract(cap100).max(0)
    drenagem = sw100_tmp.min(cap100).subtract(cc100).max(0).multiply(0.04)
    runoff_sub = excesso.add(drenagem).rename('runoff_sub_mm')
    # clamp() só aceita números; com limite por pixel (imagem) usa-se max/min
    novo_sw100 = sw100_tmp.subtract(runoff_sub).max(0).min(cap100)
    novo_sw5 = sw5.add(infiltracao).subtract(evaporacao).max(0).min(cap5)

    # tsolo/tfolha mascarados: o GFS não tem equivalente direto às bandas do ERA5,
    # e usar Tmédia/Tmin criava degrau artificial na transição ERA5 -> GFS.
    mascarado = ee.Image.constant(0).updateMask(0)

    bandas_ok = {
        'pr_mm': pr, 'tmax_C': tmax, 'tmin_C': tmin, 'RH': rh,
        'et_potencial': et_potencial, 'et_real': et_real, 'vpd': vpd,
        'rs': rs, 'u2': u2,
        'tsolo_C': mascarado.rename('tsolo_C'), 'tfolha_C': mascarado.rename('tfolha_C'),
        'transpiracao_mm': transp_b, 'runoff_sup_mm': runoff_sup, 'runoff_sub_mm': runoff_sub,
        'soil_moisture_0_5cm_m3_m3': novo_sw5.divide(50).rename('soil_moisture_0_5cm_m3_m3'),
        'soil_moisture_0_100cm_m3_m3': novo_sw100.divide(1000).rename('soil_moisture_0_100cm_m3_m3'),
    }
    img_ok = ee.Image.cat([bandas_ok[b] for b in ORDEM_BANDAS]).toFloat() \
        .resample('bilinear').reproject(crs=crsNativo, crsTransform=transformNativo)
    img_ok = expandir_borda(img_ok)

    bandas_falha = {b: mascarado.rename(b) for b in ORDEM_BANDAS}
    bandas_falha['soil_moisture_0_5cm_m3_m3'] = sw5.divide(50).rename('soil_moisture_0_5cm_m3_m3')
    bandas_falha['soil_moisture_0_100cm_m3_m3'] = sw100.divide(1000).rename('soil_moisture_0_100cm_m3_m3')
    img_falha = ee.Image.cat([bandas_falha[b] for b in ORDEM_BANDAS]).toFloat()

    registro_gee = ee.Algorithms.If(
        qtd_img.gt(0),
        ee.Date(ee.Number(gfs_dia.aggregate_max('system:version')).divide(1000)).format('YYYY-MM-dd HH:mm:ss'),
        'N/A')

    comuns = {
        'system:time_start': data_atual.millis(),
        'versao_algoritmo': VERSAO_ALGORITMO,
        'rodada_gfs_ms': creation_ms,
        'Rodada_Utilizada_NOAA': datetime.fromtimestamp(creation_ms / 1000, tz=timezone.utc).strftime('%Y-%m-%d %H:%M:%S'),
        'Imagens_Esperadas_GFS': n_img_esperado,
        'Momento_Extracao_Local': data_extracao_local,
    }
    final_ok = finalizar(img_ok, dict(comuns, versao=versao_ok, Imagens_Validas_GFS=qtd_img,
                                      Registro_Servidor_GEE=registro_gee))
    final_falha = finalizar(img_falha, dict(comuns, versao='falha_latencia_gfs', Imagens_Validas_GFS=0,
                                            Registro_Servidor_GEE='N/A'))
    return ee.Image(ee.Algorithms.If(qtd_img.gt(0), final_ok, final_falha))

def status_dia(d_str, info):
    if info is None:
        return 'ausente'
    v, alg = info['versao'], info['alg']
    if v in VERSOES_GFS:
        return 'precisa_upgrade' if d_str <= ultima_data_era5 else 'gravado'
    if v == 'provisoria_era5':
        if d_str <= ultima_data_brd:
            return 'precisa_upgrade'      # o BR-DWGD já cobre o dia: vira definitivo
        if alg != VERSAO_ALGORITMO:
            return 'precisa_upgrade'      # provisórios acompanham sempre a versão atual
        return 'gravado'
    # Dias definitivos: só mudam por decisão explícita (REPROCESSAR_VERSAO_ANTIGA ou patch)
    if REPROCESSAR_VERSAO_ANTIGA and alg != VERSAO_ALGORITMO:
        return 'precisa_upgrade'
    if alg not in ALGORITMOS_DEFINITIVOS_VALIDOS:
        return 'precisa_upgrade'          # dias da V1.1 (fórmula de calibração diferente)
    if alg == 'V1.0' and APLICAR_PATCH_SOLO:
        return 'precisa_patch'
    return 'gravado'

def exportar(img, d_str, usar_tmp):
    global capacidade
    nome = id_dia(d_str)
    desc = desc_dia(d_str)
    destino = f'{ASSET_TEMP if usar_tmp else ASSET_DESTINO}/{nome}'
    task = ee.batch.Export.image.toAsset(
        image=img, description=desc, assetId=destino,
        region=limitesRegiao, crs=crsNativo, crsTransform=transformNativo, maxPixels=1e13)
    task.start()
    tarefas[desc] = task.id
    capacidade -= 1
    return task

def aguardar_tarefa(desc):
    global capacidade
    task_id = tarefas.get(desc)
    if task_id is None:
        return 'COMPLETED'
    limite = time.time() + TIMEOUT_TAREFA_MIN * 60
    while True:
        estado = ee.data.getTaskStatus(task_id)[0].get('state')
        if estado in ESTADOS_FINAIS:
            tarefas.pop(desc, None)
            capacidade += 1
            return estado
        if time.time() > limite:
            raise TimeoutError(f"Tarefa {desc} excedeu {TIMEOUT_TAREFA_MIN} min (estado: {estado}).")
        time.sleep(INTERVALO_POLL_S)

def aguardar_dia_pronto(d_str):
    nome, desc = id_dia(d_str), desc_dia(d_str)
    if desc in tarefas:
        print(f"    ⏳ Aguardando tarefa do dia base {d_str}...")
        estado = aguardar_tarefa(desc)
        if estado != 'COMPLETED':
            raise RuntimeError(f"Dia base {d_str} terminou com estado {estado}; cadeia GFS interrompida.")
    promover_tmp(nome)
    if not asset_existe(f'{ASSET_DESTINO}/{nome}'):
        raise RuntimeError(f"Dia base {d_str} não existe no acervo e não há tarefa ativa para ele.")

def despachar_assincrono(datas, rotulo, reserva):
    global capacidade
    print(f"🔎 {rotulo}: {len(datas):,} dias avaliados.")
    enviados, por_ano = [], {}
    for d in datas:
        st_ano = por_ano.setdefault(d[:4], {'gravado': 0, 'fila': 0, 'enviado': 0, 'adiado': 0})
        nome, desc = id_dia(d), desc_dia(d)
        if desc in tarefas:
            st_ano['fila'] += 1
            continue
        status = status_dia(d, inventario.get(nome))
        if status == 'gravado':
            st_ano['gravado'] += 1
            continue
        if capacidade <= reserva:
            st_ano['adiado'] += 1
            continue
        if SIMULAR:
            info = inventario.get(nome)
            rota = 'definitiva' if d <= ultima_data_brd else 'provisoria_era5'
            if status == 'precisa_patch':
                motivo = f"patch solo/escoamento ({info['versao']}, V1.0 -> {VERSAO_PATCH})"
            elif info is None:
                motivo = f"criar ({rota})"
            else:
                motivo = f"recalcular {info['versao']} {info['alg']} -> {rota} {VERSAO_ALGORITMO}"
            relatorio_sim[motivo] = relatorio_sim.get(motivo, 0) + 1
            capacidade -= 1
            st_ano['enviado'] += 1
            enviados.append(d)
            continue
        if status == 'precisa_patch':
            img = aplicarPatch_Solo(d)
        else:
            img = processarDia_Definitivo(d) if d <= ultima_data_brd else processarDia_Rota1(d)
        exportar(img, d, usar_tmp=(status != 'ausente'))
        st_ano['enviado'] += 1
        enviados.append(d)
    for ano, s in sorted(por_ano.items()):
        if s['enviado'] or s['adiado'] or s['fila']:
            print(f"  📅 [{ano}] gravados {s['gravado']:,} | na fila {s['fila']:,} | "
                  f"enviados {s['enviado']:,} | adiados por capacidade {s['adiado']:,}")
    return enviados

def descobrir_rodadas_gfs():
    # A busca precisa cobrir todo o intervalo sem ERA5, senão dias passados ficam sem rodada
    lacuna = (data_hoje_dt.date() - datetime.strptime(ultima_data_era5, '%Y-%m-%d').date()).days
    dias_busca = max(DIAS_BUSCA_RODADAS, lacuna + 3)
    ini = (data_hoje_dt - timedelta(days=dias_busca)).strftime('%Y-%m-%d')
    fim = (data_hoje_dt + timedelta(days=1)).strftime('%Y-%m-%d')
    rodadas = ee.ImageCollection(COL_GFS).filterDate(ini, fim) \
        .aggregate_array('creation_time').distinct().getInfo()
    rodadas = sorted(int(float(r)) for r in rodadas)
    if not rodadas:
        raise RuntimeError("Nenhuma rodada GFS encontrada no período de busca.")

    referencia = None
    for r in reversed(rodadas[-4:]):
        stats = ee.Dictionary({
            'n': colecao_gfs_rodada(r).size(),
            'max': colecao_gfs_rodada(r).aggregate_max('forecast_hours'),
        }).getInfo()
        if stats.get('max') is not None and stats['max'] >= 384 and stats['n'] >= 198:
            referencia = r
            break
    if referencia is None:
        referencia = rodadas[-1]
        print("  ⚠️ Nenhuma das últimas rodadas está completa; usando a mais recente.")
    print(f"  🛰️ Rodada GFS de referência: "
          f"{datetime.fromtimestamp(referencia / 1000, tz=timezone.utc).strftime('%Y-%m-%d %HZ')}")
    return rodadas, referencia

def rodada_para_dia(d_str, rodadas, referencia):
    t0 = data_para_ms(d_str)
    if t0 >= referencia:
        return referencia
    candidatas = [r for r in rodadas if t0 - 48 * 3600000 <= r <= t0 and r <= referencia]
    return max(candidatas) if candidatas else referencia

def referencia_balde(ultima_era5):
    """Razões ET real/ETp e transpiração/ET real dos últimos DIAS_REF_ET dias ERA5,
    e a umidade do último dia ERA5, por pixel."""
    ini = somar_dias(ultima_era5, -(DIAS_REF_ET - 1))
    col = ee.ImageCollection(ASSET_DESTINO).filterDate(ini, somar_dias(ultima_era5, 1))
    et_r = col.select('et_real').mean()
    et_p = col.select('et_potencial').mean()
    tr = col.select('transpiracao_mm').mean()
    base = ee.Image(f'{ASSET_DESTINO}/{id_dia(ultima_era5)}')
    return ee.Image.cat([
        et_r.divide(et_p.max(0.1)).clamp(0.05, 1.0).rename('k_et'),
        tr.divide(et_r.max(0.05)).clamp(0.0, 1.0).rename('f_transp'),
        base.select('soil_moisture_0_100cm_m3_m3').multiply(1000).rename('sw_ref_100'),
        base.select('soil_moisture_0_5cm_m3_m3').multiply(50).rename('sw_ref_5'),
    ])

def executar_cadeia_gfs(enviados_operacionais):
    dias = intervalo_datas(somar_dias(ultima_data_era5, 1), data_fim_processo)
    if not dias:
        return
    print(f"\n⚙ Cadeia GFS sequencial: {dias[0]} a {dias[-1]} ({len(dias)} dias).")
    rodadas, referencia = descobrir_rodadas_gfs()

    dias_ref = intervalo_datas(somar_dias(ultima_data_era5, -(DIAS_REF_ET - 1)), ultima_data_era5)
    ref = None
    if not SIMULAR:
        for d_ref in dias_ref:            # a referência precisa estar com as versões finais
            aguardar_dia_pronto(d_ref)
        ref = referencia_balde(ultima_data_era5)
    forcar = any(d_ref in enviados_operacionais for d_ref in dias_ref)

    for d in dias:
        nome, desc = id_dia(d), desc_dia(d)
        caminho = f'{ASSET_DESTINO}/{nome}'
        if desc in tarefas and not SIMULAR:
            aguardar_tarefa(desc)
            promover_tmp(nome)

        rodada = rodada_para_dia(d, rodadas, referencia)
        info = info_de_propriedades(propriedades_asset(caminho))
        precisa = (forcar or info is None or info['rodada'] != rodada
                   or info['alg'] != VERSAO_ALGORITMO)
        if not precisa:
            print(f"    ✔️ {d}: atualizado com a rodada esperada.")
            continue
        forcar = True   # estado do solo mudou: todos os dias seguintes precisam ser refeitos
        if SIMULAR:
            motivo = 'criar (GFS)' if info is None else f"refazer {info['versao']} -> GFS"
            relatorio_sim[motivo] = relatorio_sim.get(motivo, 0) + 1
            continue

        d_ant = somar_dias(d, -1)
        aguardar_dia_pronto(d_ant)
        base = ee.Image(f'{ASSET_DESTINO}/{id_dia(d_ant)}')
        estado = ee.Image.cat([
            base.select('soil_moisture_0_100cm_m3_m3').multiply(1000).rename('sw_100'),
            base.select('soil_moisture_0_5cm_m3_m3').multiply(50).rename('sw_5'),
        ])

        n_img, horas_chuva = contagem_esperada(data_para_ms(d), rodada)
        img = processarDia_GFS(d, estado, ref, rodada, n_img, horas_chuva)
        usar_tmp = info is not None
        exportar(img, d, usar_tmp)
        estado_final = aguardar_tarefa(desc)
        if estado_final != 'COMPLETED':
            raise RuntimeError(f"Falha na cadeia GFS: {desc} terminou com {estado_final}.")
        if usar_tmp:
            promover_tmp(nome)
        print(f"    ✔️ {d}: exportado ({n_img} imagens esperadas, {len(horas_chuva)} blocos de chuva).")

# Estados de cada região com os nomes usados pelo FAO/GAUL 2015 (sem acentos)
UFS_REGIAO = {
    'Norte': ['Acre', 'Amapa', 'Amazonas', 'Para', 'Rondonia', 'Roraima', 'Tocantins'],
    'Nordeste': ['Alagoas', 'Bahia', 'Ceara', 'Maranhao', 'Paraiba', 'Pernambuco', 'Piaui',
                 'Rio Grande Do Norte', 'Sergipe'],
    'Centro-Oeste': ['Distrito Federal', 'Goias', 'Mato Grosso', 'Mato Grosso Do Sul'],
    'Sudeste': ['Espirito Santo', 'Minas Gerais', 'Rio De Janeiro', 'Sao Paulo'],
    'Sul': ['Parana', 'Rio Grande Do Sul', 'Santa Catarina'],
}
CODIGO_REGIAO = {reg: k for k, reg in enumerate(UFS_REGIAO, start=1)}   # 0 = pixel fora das UFs listadas
BANDAS_PREVIA = list(ORDEM_BANDAS)   # as 16 bandas do SAB
# O que a V1.3 faz com cada banda dos dias provisórios, pelo código (V1.0 x V1.3):
BANDAS_MUDAM = ('et_potencial', 'et_real', 'transpiracao_mm')   # ETo nova; ET real e transpiração
#                 têm teto (ET real <= ETo, transpiração <= ET real), então acompanham a ETo nova
BANDAS_FATOR = ('pr_mm', 'u2', 'rs')   # calibração multiplicativa: só mudam onde o teto FATOR_MAX atua
BANDAS_SOLO = ('soil_moisture_0_5cm_m3_m3', 'soil_moisture_0_100cm_m3_m3',
               'runoff_sup_mm', 'runoff_sub_mm')   # o painel já corrige os dias da V1.0 na leitura
BANDAS_IGUAIS = ('tmax_C', 'tmin_C', 'RH', 'vpd', 'tsolo_C', 'tfolha_C')   # mesma conta, mesmos dados
assert sorted(BANDAS_MUDAM + BANDAS_FATOR + BANDAS_SOLO + BANDAS_IGUAIS) == sorted(BANDAS_PREVIA)
GRUPO_BANDA = {**{b: 'muda na V1.3' for b in BANDAS_MUDAM}, **{b: 'teto do fator' for b in BANDAS_FATOR},
               **{b: 'solo/escoamento' for b in BANDAS_SOLO}, **{b: 'deve ficar igual' for b in BANDAS_IGUAIS}}
UNIDADE_BANDA = {'pr_mm': 'mm', 'tmax_C': '°C', 'tmin_C': '°C', 'RH': '%', 'et_potencial': 'mm',
                 'et_real': 'mm', 'vpd': 'kPa', 'rs': 'MJ/m²', 'u2': 'm/s', 'tsolo_C': '°C',
                 'tfolha_C': '°C', 'transpiracao_mm': 'mm', 'runoff_sup_mm': 'mm', 'runoff_sub_mm': 'mm',
                 'soil_moisture_0_5cm_m3_m3': 'm³/m³', 'soil_moisture_0_100cm_m3_m3': 'm³/m³'}
# Diferença mínima, na unidade da banda, para um pixel contar como alterado
TOLERANCIA_PREVIA = {b: 0.01 for b in BANDAS_PREVIA}
TOLERANCIA_PREVIA.update({'RH': 0.1, 'vpd': 0.001, 'soil_moisture_0_5cm_m3_m3': 0.001,
                          'soil_moisture_0_100cm_m3_m3': 0.001})
# Coerência física de cada imagem (fração de pixels com o problema)
COERENCIA = {
    'etr_acima_eto': 'ET real acima da ETo',
    'transp_acima_etr': 'transpiração acima da ET real',
    'tmax_abaixo_tmin': 'Tmax abaixo da Tmin',
    'ur_fora': 'UR fora de 0 a 100%',
    'umid_acima_teto': f'umidade do solo acima de {UMIDADE_MAX_FISICA}',
}
# Bandas somadas pela prévia, na ordem do redutor: a_ = como o painel mostra hoje, n_ = V1.3,
# d_ = |V1.3 - hoje|, f_ = 1 se o pixel mudou mais que a tolerância. n_pix vale 1 em cada pixel
# válido: a soma dele é a contagem de pixels, e média = soma da banda / soma de n_pix.
NOMES_SOMA = ([f'{p}_{b}' for p in ('a', 'n', 'd', 'f') for b in BANDAS_PREVIA]
              + ['a_deficit', 'n_deficit', 'clim_eto', 'f_solo']
              + [f'{p}_{c}' for p in ('a', 'n') for c in COERENCIA] + ['n_pix'])
# Tentativas de cálculo de cada dia: (n, tileScale) = retângulo do Brasil cortado em n x n partes.
# Se o Earth Engine recusar por memória ou tempo, passa para a tentativa seguinte, com partes
# menores; as somas das partes fecham exatamente com a do retângulo inteiro.
NIVEIS_PREVIA = ((1, 4), (2, 16), (4, 16))

# Critérios de aprovação, definidos antes de rodar. No teste E4 (953 pontos, 2016-2025), a ETo
# atual ficou entre +6,5% e +10,2% acima do BR-DWGD e a de Penman-Monteith entre +1,8% e +3,5%;
# a troca deve, portanto, reduzir a ETo nacional em algo como 4% a 7%. As bandas fora de
# BANDAS_MUDAM não deveriam mudar, a não ser nos poucos pixels em que o teto do fator de
# calibração (FATOR_MAX, da V1.2) atua.
CRIT_ETO_BRASIL = (-12.0, -1.0)      # variação aceitável da ETo nacional (%)
CRIT_ETO_REGIAO = (-20.0, 5.0)       # variação aceitável da ETo de cada região (%)
CRIT_MAX_PIXELS_FATOR = 2.0          # % máxima de pixels alterados em chuva, vento e radiação
CRIT_MAX_PIXELS_SOLO = 1.0           # % máxima de pixels com umidade ou escoamento alterados
CRIT_MAX_PIXELS_IGUAIS = 0.5         # % máxima de pixels alterados nas bandas que devem ficar iguais
CRIT_MAX_INCOERENCIA = 0.01          # % máxima de pixels incoerentes na V1.3 (pelo código, deve ser 0)
CRIT_MIN_DIAS = 2 / 3                # fração mínima dos dias amostrados que precisa ser calculada
CRIT_MIN_COBERTURA = 0.95            # as cinco regiões juntas precisam ter 95% dos pixels do Brasil

def _imagem_como_no_painel(d):
    """O dia gravado como o painel o mostra hoje: nos dias da V1.0 o painel aplica, na leitura,
    a correção de umidade e escoamento, que é a mesma do aplicarPatch_Solo."""
    info = inventario.get(id_dia(d))
    if info is not None and info['alg'] == 'V1.0':
        return aplicarPatch_Solo(d)
    return ee.Image(f'{ASSET_DESTINO}/{id_dia(d)}')

def _rotulo_regioes():
    """Imagem com o código da região de cada pixel (1 a 5, na ordem de UFS_REGIAO; 0 = pixel fora
    das UFs listadas). Os estados são pintados na grade, sem unir polígonos: a união
    (FeatureCollection.geometry) é a operação que o Earth Engine desaconselha por consumir muita
    memória, e era feita a cada dia na versão anterior da prévia. Usa a versão do GAUL 2015
    simplificada a 500 m (mesmos nomes e atributos), mais leve e suficiente para pixels de 11 km;
    o recorte do Brasil continua vindo das próprias imagens do SAB."""
    estados = (ee.FeatureCollection('FAO/GAUL_SIMPLIFIED_500m/2015/level1')
               .filter(ee.Filter.eq('ADM0_NAME', 'Brazil')))
    nomes_gaul = set(estados.aggregate_array('ADM1_NAME').getInfo())
    rotulo = ee.Image.constant(0).toByte()
    for reg, ufs in UFS_REGIAO.items():
        faltam = [u for u in ufs if u not in nomes_gaul]
        if faltam:
            print(f"   ⚠️ {reg}: estados não encontrados no GAUL, fora da prévia: {faltam}")
        rotulo = rotulo.paint(estados.filter(ee.Filter.inList('ADM1_NAME', ufs)), CODIGO_REGIAO[reg])
    return rotulo.rename('cod_regiao')

def _retangulos_previa(limites, n):
    """Retângulo do Brasil cortado em n x n partes, com os cortes sobre as linhas da grade de 0,1°
    do SAB: cada pixel cai em uma única parte. limites = (oeste, sul, leste, norte), em graus."""
    oeste, sul, leste, norte = limites
    dx, x0 = transformNativo[0], transformNativo[2]
    dy, y0 = -transformNativo[4], transformNativo[5]
    c0, c1 = math.floor((oeste - x0) / dx), math.ceil((leste - x0) / dx)
    l0, l1 = math.floor((y0 - norte) / dy), math.ceil((y0 - sul) / dy)
    cols = [c0 + round(i * (c1 - c0) / n) for i in range(n + 1)]
    lins = [l0 + round(i * (l1 - l0) / n) for i in range(n + 1)]
    return [[x0 + ca * dx, y0 - lb * dy, x0 + cb * dx, y0 - la * dy]
            for la, lb in zip(lins, lins[1:]) for ca, cb in zip(cols, cols[1:])]

def _coerencia(img):
    """Marca, pixel a pixel, os problemas físicos listados em COERENCIA (1 = problema)."""
    teto = UMIDADE_MAX_FISICA + 1e-4
    return ee.Image.cat([
        img.select('et_real').gt(img.select('et_potencial').add(0.001)),
        img.select('transpiracao_mm').gt(img.select('et_real').add(0.001)),
        img.select('tmax_C').lt(img.select('tmin_C')),
        img.select('RH').lt(0).Or(img.select('RH').gt(100)),
        img.select('soil_moisture_0_100cm_m3_m3').gt(teto).Or(img.select('soil_moisture_0_5cm_m3_m3').gt(teto)),
    ]).rename(list(COERENCIA))

def _pilha_previa(d, rotulo):
    """As 16 bandas do dia como estão hoje (a_*) e como a V1.3 gravaria (n_*), a diferença (d_*),
    a marca de pixel alterado (f_*), déficits, ETo climatológica do mês, coerência física, n_pix e,
    por último, o código da região."""
    antigo = _imagem_como_no_painel(d).select(BANDAS_PREVIA)
    novo = processarDia_Rota1(d).select(BANDAS_PREVIA)
    clima_eto = expandir_borda(ee.Image(f'{ASSET_CLIMATOLOGIA}/Matriz_Clima_{d[5:7]}').select('ET_m_br'))

    def deficit(img):
        return img.select('et_potencial').subtract(img.select('et_real'))

    dif = novo.subtract(antigo).abs().rename(BANDAS_PREVIA)      # banda a banda, na mesma ordem
    mudou = dif.gt(ee.Image.constant([TOLERANCIA_PREVIA[b] for b in BANDAS_PREVIA])).rename(BANDAS_PREVIA)

    pilha = ee.Image.cat([
        antigo.rename([f'a_{b}' for b in BANDAS_PREVIA]),
        novo.rename([f'n_{b}' for b in BANDAS_PREVIA]),
        dif.rename([f'd_{b}' for b in BANDAS_PREVIA]),
        mudou.rename([f'f_{b}' for b in BANDAS_PREVIA]),
        deficit(antigo).rename('a_deficit'),
        deficit(novo).rename('n_deficit'),
        clima_eto.rename('clim_eto'),
        mudou.select(list(BANDAS_SOLO)).reduce(ee.Reducer.max()).rename('f_solo'),
        _coerencia(antigo).rename([f'a_{c}' for c in COERENCIA]),
        _coerencia(novo).rename([f'n_{c}' for c in COERENCIA]),
        ee.Image.constant(1).rename('n_pix'),
        rotulo,
    ]).select(NOMES_SOMA + ['cod_regiao'])
    # Uma só máscara para todas as bandas: hoje e V1.3 são comparados exatamente nos mesmos pixels.
    # Fora do Brasil as imagens do SAB são mascaradas, então o retângulo não soma oceano nem vizinhos.
    return pilha.updateMask(pilha.mask().reduce(ee.Reducer.min()))

class _PedidoGrande(Exception):
    """O Earth Engine recusou o pedido por memória ou por tempo."""

def _pedido_grande_demais(erro):
    msg = str(erro).lower()
    return 'memory' in msg or 'timed out' in msg

def _somas_regiao(pilha, retangulos, tile_scale):
    """Soma de cada banda de NOMES_SOMA por código de região, parte por parte (um pedido por parte)."""
    n = len(NOMES_SOMA)
    redutor = ee.Reducer.sum().repeat(n).group(groupField=n, groupName='cod_regiao')
    total = {}
    for ret in retangulos:
        geometria = ee.Geometry.Rectangle(ret, proj=crsNativo, geodesic=False)
        for tentativa in range(3):
            try:
                r = pilha.reduceRegion(reducer=redutor, geometry=geometria, crs=crsNativo,
                                       crsTransform=transformNativo, tileScale=tile_scale,
                                       maxPixels=1e10).getInfo()
                break
            except Exception as e:
                if _pedido_grande_demais(e):
                    raise _PedidoGrande(str(e)) from None
                if tentativa == 2:
                    raise
                time.sleep(15)   # falha passageira do servidor (ex.: 503 ou excesso de pedidos)
        for grupo in r.get('groups', []):
            cod = int(grupo['cod_regiao'])
            anterior = total.get(cod, [0.0] * n)
            total[cod] = [a + (s or 0.0) for a, s in zip(anterior, grupo['sum'])]
    return total

def _medias_previa(somas):
    """Converte as somas por código em médias por região; o Brasil soma todos os códigos, inclusive
    o 0. Região sem pixel nenhum fica de fora (e o resumo acusa a ausência)."""
    i_pix = NOMES_SOMA.index('n_pix')
    nome = {k: reg for reg, k in CODIGO_REGIAO.items()}

    def medias(v):
        saida = {b: v[i] / v[i_pix] for i, b in enumerate(NOMES_SOMA) if b != 'n_pix'}
        saida['n_pix'] = v[i_pix]
        return saida

    resultado = {nome[k]: medias(v) for k, v in somas.items() if k in nome and v[i_pix] > 0}
    brasil = [sum(col) for col in zip(*somas.values())]
    if brasil and brasil[i_pix] > 0:
        resultado['Brasil'] = medias(brasil)
    return resultado

def _previa_dia(d, rotulo, limites):
    """Médias regionais do dia como está hoje (a_*) e como a V1.3 gravaria (n_*). Se o Earth Engine
    recusar por memória ou tempo, refaz o mesmo cálculo em partes menores (NIVEIS_PREVIA)."""
    pilha = _pilha_previa(d, rotulo)
    for k, (n, tile_scale) in enumerate(NIVEIS_PREVIA):
        try:
            resultado = _medias_previa(_somas_regiao(pilha, _retangulos_previa(limites, n), tile_scale))
            if 'Brasil' not in resultado:
                raise RuntimeError("nenhum pixel válido em comum entre o dia gravado e a V1.3")
            return resultado
        except _PedidoGrande as e:
            if k == len(NIVEIS_PREVIA) - 1:
                raise RuntimeError(f"{e} Recusado mesmo em {n * n} partes.") from None
            prox = NIVEIS_PREVIA[k + 1][0] ** 2
            print(f"   {d}: o Earth Engine recusou por memória ou tempo; refazendo em {prox} partes...")

def _resumir_previa(df, motivos_simulacao, dias_previstos=None):
    """Monta as tabelas da prévia e aplica os critérios. Não usa o Earth Engine."""
    ordem = [r for r in ['Brasil', *UFS_REGIAO] if r in set(df['regiao'])]
    br = df[df['regiao'] == 'Brasil'].set_index('data').sort_index()
    por_data = pd.DataFrame({
        'ETo hoje': br['a_et_potencial'], 'ETo V1.3': br['n_et_potencial'],
        'ΔETo %': 100 * (br['n_et_potencial'] / br['a_et_potencial'] - 1),
        'ETo clima do mês': br['clim_eto'],
        'ET real hoje': br['a_et_real'], 'ET real V1.3': br['n_et_real'],
        'Déficit hoje': br['a_deficit'], 'Déficit V1.3': br['n_deficit'],
    }).astype(float).round(2)

    g = df.groupby('regiao')
    soma = lambda col: g[col].sum()
    variacao = lambda a, n: 100 * (soma(n) / soma(a) - 1)
    por_regiao = pd.DataFrame({
        'ΔETo %': variacao('a_et_potencial', 'n_et_potencial'),
        'ΔET real %': variacao('a_et_real', 'n_et_real'),
        'ΔDéficit %': variacao('a_deficit', 'n_deficit'),
        'ETo hoje x clima %': 100 * (soma('a_et_potencial') / soma('clim_eto') - 1),
        'ETo V1.3 x clima %': 100 * (soma('n_et_potencial') / soma('clim_eto') - 1),
        'ΔChuva %': variacao('a_pr_mm', 'n_pr_mm'),
        '% pixels chuva alterada': 100 * g['f_pr_mm'].mean(),
        '% pixels solo alterado': 100 * g['f_solo'].mean(),
    }).reindex(ordem).astype(float).round(2)

    # As 16 bandas no Brasil: média dos dias amostrados, hoje x V1.3
    hoje = pd.Series({b: br[f'a_{b}'].mean() for b in BANDAS_PREVIA}, dtype=float)
    v13 = pd.Series({b: br[f'n_{b}'].mean() for b in BANDAS_PREVIA}, dtype=float)
    n_alterados = pd.Series({b: br[f'f_{b}'].sum() for b in BANDAS_PREVIA}, dtype=float)
    todas = pd.DataFrame({
        'grupo': [GRUPO_BANDA[b] for b in BANDAS_PREVIA],
        'unid.': [UNIDADE_BANDA[b] for b in BANDAS_PREVIA],
        'hoje': hoje, 'V1.3': v13, 'Δ': v13 - hoje,
        'Δ %': 100 * (v13 / hoje.where(hoje.abs() > 1e-9) - 1),
        '% pixels alterados': pd.Series({b: 100 * br[f'f_{b}'].mean() for b in BANDAS_PREVIA}, dtype=float),
        '|Δ| onde mudou': pd.Series({b: br[f'd_{b}'].sum() / n_alterados[b] if n_alterados[b] > 0 else 0.0
                                     for b in BANDAS_PREVIA}, dtype=float),
    }, index=pd.Index(BANDAS_PREVIA, name='banda'))
    num = todas.columns[2:]
    todas[num] = todas[num].astype(float).round(3)

    coerencia = pd.DataFrame({
        'hoje (% pixels)': [100 * br[f'a_{c}'].mean() for c in COERENCIA],
        'V1.3 (% pixels)': [100 * br[f'n_{c}'].mean() for c in COERENCIA],
    }, index=pd.Index(list(COERENCIA.values()), name='verificação')).astype(float).round(3)

    falhas = []
    definitivos = [m for m in motivos_simulacao
                   if m.startswith('recalcular definitiva') or m.startswith('patch')]
    if definitivos:
        falhas.append(f"a execução mexeria em dias definitivos: {definitivos}")
    if 'Brasil' in por_regiao.index:
        b = por_regiao.loc['Brasil']
        if not CRIT_ETO_BRASIL[0] <= b['ΔETo %'] <= CRIT_ETO_BRASIL[1]:
            falhas.append(f"ETo nacional variou {b['ΔETo %']:+.1f}% "
                          f"(esperado entre {CRIT_ETO_BRASIL[0]:+.0f}% e {CRIT_ETO_BRASIL[1]:+.0f}%)")
        for banda in BANDAS_FATOR:
            p = todas.loc[banda, '% pixels alterados']
            if p > CRIT_MAX_PIXELS_FATOR:
                falhas.append(f"{banda} alterada em {p:.2f}% dos pixels (limite {CRIT_MAX_PIXELS_FATOR:.0f}%)")
        if b['% pixels solo alterado'] > CRIT_MAX_PIXELS_SOLO:
            falhas.append(f"umidade ou escoamento alterados em {b['% pixels solo alterado']:.2f}% "
                          f"dos pixels (limite {CRIT_MAX_PIXELS_SOLO:.0f}%)")
        for banda in BANDAS_IGUAIS:
            p = todas.loc[banda, '% pixels alterados']
            if p > CRIT_MAX_PIXELS_IGUAIS:
                falhas.append(f"{banda} mudou em {p:.2f}% dos pixels, mas a conta é a mesma da versão "
                              f"atual (limite {CRIT_MAX_PIXELS_IGUAIS}%)")
        for chave, texto in COERENCIA.items():
            agora, novo = coerencia.loc[texto, 'hoje (% pixels)'], coerencia.loc[texto, 'V1.3 (% pixels)']
            # Tmax < Tmin vem da calibração aditiva, igual nas duas versões: a V1.3 só não pode piorar
            limite = agora + CRIT_MAX_INCOERENCIA if chave == 'tmax_abaixo_tmin' else CRIT_MAX_INCOERENCIA
            if novo > limite:
                falhas.append(f"V1.3 com {texto} em {novo:.3f}% dos pixels (limite {limite:.3f}%)")
    else:
        falhas.append("a prévia não conseguiu calcular a média do Brasil")
    for r in UFS_REGIAO:
        if r not in por_regiao.index:
            falhas.append(f"a prévia não conseguiu calcular a região {r}")
            continue
        v = por_regiao.loc[r, 'ΔETo %']
        if not CRIT_ETO_REGIAO[0] <= v <= CRIT_ETO_REGIAO[1]:
            falhas.append(f"ETo do {r} variou {v:+.1f}% "
                          f"(esperado entre {CRIT_ETO_REGIAO[0]:+.0f}% e {CRIT_ETO_REGIAO[1]:+.0f}%)")
    # Controles da própria prévia: dias suficientes e regiões cobrindo o país
    n_dias = df['data'].nunique()
    if dias_previstos and n_dias < math.ceil(CRIT_MIN_DIAS * dias_previstos):
        falhas.append(f"só {n_dias} de {dias_previstos} dias foram calculados "
                      f"(mínimo {math.ceil(CRIT_MIN_DIAS * dias_previstos)})")
    if 'n_pix' in df.columns and 'Brasil' in set(df['regiao']):
        pix = df.pivot_table(index='data', columns='regiao', values='n_pix', aggfunc='sum')
        regs = [r for r in UFS_REGIAO if r in pix.columns]
        cobertura = (pix[regs].sum(axis=1) / pix['Brasil']).min() if regs else 0.0
        if cobertura < CRIT_MIN_COBERTURA:
            falhas.append(f"as regiões cobrem só {100 * cobertura:.0f}% dos pixels do Brasil "
                          f"(mínimo {100 * CRIT_MIN_COBERTURA:.0f}%)")
    return por_data, por_regiao, todas, coerencia, falhas

def gerar_matriz_do_mes(mes):
    br_hist = colChuvaGeral.filter(ee.Filter.calendarRange(mes, mes, 'month')).filterDate('1986-01-01', '2026-01-01')
    era_hist = (ee.ImageCollection('ECMWF/ERA5_LAND/DAILY_AGGR')
                .filter(ee.Filter.calendarRange(mes, mes, 'month'))
                .filterDate('1986-01-01', '2026-01-01')
                .filterBounds(limitesRegiao))

    def preparar_era(img):
        pr = img.select('total_precipitation_sum').multiply(1000).max(0).rename('pr')
        tmax = img.select('temperature_2m_max').subtract(273.15).resample('bilinear').rename('Tmax')
        tmin = img.select('temperature_2m_min').subtract(273.15).resample('bilinear').rename('Tmin')
        et = img.select('potential_evaporation_sum').multiply(-1000).max(0).resample('bilinear').rename('ET')
        rs = img.select('surface_solar_radiation_downwards_sum').divide(1000000).resample('bilinear').rename('Rs')
        u2 = img.select('u_component_of_wind_10m').hypot(img.select('v_component_of_wind_10m')).multiply(0.748).resample('bilinear').rename('u2')

        t2m = img.select('temperature_2m').subtract(273.15).resample('bilinear')
        tdew = img.select('dewpoint_temperature_2m').subtract(273.15).resample('bilinear')
        rh = t2m.expression('100 * (exp((17.625 * Td) / (243.04 + Td)) / exp((17.625 * T) / (243.04 + T)))', {'T': t2m, 'Td': tdew}).rename('RH')

        return ee.Image.cat([pr, tmax, tmin, et, rs, u2, rh])

    def limpar_br(img):
        mask = img.select('pr').gt(-32768)
        return ee.Image.cat([
            img.select('pr').updateMask(mask).multiply(0.00686666).add(225.0).max(0).rename('pr'),
            img.select('Tmax').updateMask(mask).multiply(0.00106815).add(15.0).rename('Tmax'),
            img.select('Tmin').updateMask(mask).multiply(0.00106815).add(15.0).rename('Tmin'),
            img.select('RH').updateMask(mask).multiply(0.39370079).add(-0.393701).rename('RH'),
            img.select('u2').updateMask(mask).multiply(0.05905512).add(-0.059055).rename('u2'),
            img.select('Rs').updateMask(mask).multiply(0.15708661).add(-0.057087).rename('Rs'),
            img.select('ET').updateMask(mask).multiply(0.05118110).add(-0.0511811).rename('ET')
        ])

    br_limpo = br_hist.map(limpar_br)
    era_limpo = era_hist.map(preparar_era)

    img_mean_br = br_limpo.mean().rename(['pr_m_br', 'Tmax_m_br', 'Tmin_m_br', 'RH_m_br', 'u2_m_br', 'Rs_m_br', 'ET_m_br'])
    img_std_br = br_limpo.reduce(ee.Reducer.stdDev()).rename(['pr_s_br', 'Tmax_s_br', 'Tmin_s_br', 'RH_s_br', 'u2_s_br', 'Rs_s_br', 'ET_s_br'])

    img_mean_era = era_limpo.mean().rename(['pr_m_era', 'Tmax_m_era', 'Tmin_m_era', 'ET_m_era', 'Rs_m_era', 'u2_m_era', 'RH_m_era'])
    img_std_era = era_limpo.reduce(ee.Reducer.stdDev()).rename(['pr_s_era', 'Tmax_s_era', 'Tmin_s_era', 'ET_s_era', 'Rs_s_era', 'u2_s_era', 'RH_s_era'])

    # Correção: Remoção total do reproject e clip desta etapa intermédia.
    return (ee.Image.cat([img_mean_br, img_std_br, img_mean_era, img_std_era])
            .toFloat()
            .set('mes_referencia', mes))

def gerar_matriz_gfs_mes(mes):
    data_inicio = '2015-08-01'
    data_fim = '2025-12-31'

    mes_filtro = ee.Filter.calendarRange(mes, mes, 'month')

    # A) BASE GABARITO (Verdade Histórica)
    sicsb_hist = ee.ImageCollection(ASSET_SICSB_REAL) \
        .filterDate(data_inicio, data_fim) \
        .filter(mes_filtro)

    def extrair_sicsb(img):
        # Seleciona as bandas corretas da base BR-DWGD e aplica os fatores de escala e offset
        pr = img.select('pr').multiply(0.00686666).add(225.0)
        tmax = img.select('Tmax').multiply(0.00106815).add(15.0)
        tmin = img.select('Tmin').multiply(0.00106815).add(15.0)
        rh = img.select('RH').multiply(0.39370079).add(-0.393701)
        u2 = img.select('u2').multiply(0.05905512).add(-0.059055)
        rs = img.select('Rs').multiply(0.15708661).add(-0.057087)
        et = img.select('ET').multiply(0.05118110).add(-0.0511811)

        return ee.Image.cat([pr, tmax, tmin, rh, u2, rs, et]) \
            .rename(['pr', 'Tmax', 'Tmin', 'RH', 'u2', 'Rs', 'ET']) \
            .set('system:time_start', img.get('system:time_start'))

    sicsb_limpo = sicsb_hist.map(extrair_sicsb)

    # B) BASE PREDITIVA BRUTA (Modelo GFS Histórico)
    gfs_raw = ee.ImageCollection('NOAA/GFS0P25') \
        .filterBounds(limitesRegiao) \
        .filterDate(data_inicio, data_fim)

    # Função de emparelhamento estrutural com blindagem contra falhas do catálogo NOAA
    def processar_gfs_emparelhado(img_sicsb):
        data_ms = img_sicsb.date().millis()

        # Isola a rodada gerada exatamente à meia-noite do dia correspondente
        rodada = gfs_raw.filter(ee.Filter.eq('creation_time', data_ms)) \
                        .filter(ee.Filter.gt('forecast_hours', 0)) \
                        .filter(ee.Filter.lte('forecast_hours', 24))

        # Trava de segurança: conta as imagens da rodada para prevenir colapso estatístico
        tamanho_rodada = rodada.size()

        def rodada_valida():
            # 1. Temperatura: Extração de máximas e mínimas sobre a pilha de 24h
            t_base = rodada.select('temperature_2m_above_ground')
            tmax_gfs = t_base.max().rename('Tmax')
            tmin_gfs = t_base.min().rename('Tmin')

            # 2. Precipitação: Soma estrita dos blocos cumulativos para fechar 24h sem dupla contagem
            chuva_blocos = rodada.filter(ee.Filter.inList('forecast_hours', [6, 12, 18, 24]))
            pr_gfs = chuva_blocos.select('total_precipitation_surface').sum().max(0).rename('pr')

            # 3. Umidade e Radiação Solar incidente
            rh_gfs = rodada.select('relative_humidity_2m_above_ground').mean().clamp(0, 100).rename('RH')
            rs_gfs = rodada.select('downward_shortwave_radiation_flux').mean().multiply(0.0864).max(0).rename('Rs')

            # 4. Vento com ajuste logarítmico FAO-56
            def calc_u2(img):
                u = img.select('u_component_of_wind_10m_above_ground')
                v = img.select('v_component_of_wind_10m_above_ground')
                u2 = u.hypot(v).multiply(0.748).rename('u2')
                return img.addBands(u2)

            u2_gfs = rodada.map(calc_u2).select('u2').mean()

            # 5. Evapotranspiração FAO-56 Hargreaves (Cálculo Astronômico de Ra)
            J = img_sicsb.date().getRelative('day', 'year').add(1).toDouble()
            lat = ee.Image.pixelLonLat().select('latitude').multiply(math.pi / 180.0)

            dr = J.multiply(2 * math.pi / 365).cos().multiply(0.033).add(1)
            delta = J.multiply(2 * math.pi / 365).subtract(1.39).sin().multiply(0.409)

            ws_arg = lat.tan().multiply(-1).multiply(delta.tan()).clamp(-1.0, 1.0)
            ws = ws_arg.acos()

            term1 = ws.multiply(lat.sin()).multiply(delta.sin())
            term2 = lat.cos().multiply(delta.cos()).multiply(ws.sin())

            # Inversão da ordem de multiplicação para respeitar a tipagem (Image x Number)
            Ra = term1.add(term2).multiply(dr).multiply((24 * 60 / math.pi) * 0.0820)

            tmedia = tmax_gfs.add(tmin_gfs).divide(2)
            et_gfs = tmedia.add(17.8).multiply(tmax_gfs.subtract(tmin_gfs).max(0).sqrt()).multiply(0.0023).multiply(Ra).rename('ET')

            return ee.Image.cat([pr_gfs, tmax_gfs, tmin_gfs, rh_gfs, u2_gfs, rs_gfs, et_gfs]) \
                .set('system:time_start', data_ms) \
                .resample('bicubic')

        def rodada_vazia():
            # Injeta imagem vazia perfeitamente nomeada e mascarada para o GEE ignorar no cálculo de média
            vazio = ee.Image.constant([0, 0, 0, 0, 0, 0, 0]) \
                .rename(['pr', 'Tmax', 'Tmin', 'RH', 'u2', 'Rs', 'ET']) \
                .updateMask(ee.Image.constant(0)) \
                .set('system:time_start', data_ms)
            return vazio

        return ee.Image(ee.Algorithms.If(tamanho_rodada.gt(0), rodada_valida(), rodada_vazia()))

    gfs_limpo = sicsb_hist.map(processar_gfs_emparelhado)

    # C) CÁLCULO ESTATÍSTICO Z-SCORE
    sicsb_mean = sicsb_limpo.mean().rename(['pr_m_sicsb', 'Tmax_m_sicsb', 'Tmin_m_sicsb', 'RH_m_sicsb', 'u2_m_sicsb', 'Rs_m_sicsb', 'ET_m_sicsb'])
    sicsb_std = sicsb_limpo.reduce(ee.Reducer.stdDev()).rename(['pr_s_sicsb', 'Tmax_s_sicsb', 'Tmin_s_sicsb', 'RH_s_sicsb', 'u2_s_sicsb', 'Rs_s_sicsb', 'ET_s_sicsb'])

    gfs_mean = gfs_limpo.mean().rename(['pr_m_gfs', 'Tmax_m_gfs', 'Tmin_m_gfs', 'RH_m_gfs', 'u2_m_gfs', 'Rs_m_gfs', 'ET_m_gfs'])
    gfs_std = gfs_limpo.reduce(ee.Reducer.stdDev()).rename(['pr_s_gfs', 'Tmax_s_gfs', 'Tmin_s_gfs', 'RH_s_gfs', 'u2_s_gfs', 'Rs_s_gfs', 'ET_s_gfs'])

    matriz_final = ee.Image.cat([sicsb_mean, sicsb_std, gfs_mean, gfs_std]).toFloat()

    mascara_brasil = ee.Image.constant(1).clipToCollection(ee.FeatureCollection([poligonoBrasil])).mask()
    return matriz_final.updateMask(mascara_brasil).set('mes_referencia', mes)

def gerar_matriz_solo_mes(mes):
    data_inicio = '2015-04-01'
    data_fim = '2026-01-01'

    # COLEÇÃO 1: SMAP L4 (A "Verdade Absoluta" do Presente)
    smap_hist = (ee.ImageCollection("NASA/SMAP/SPL4SMGP/008")
                 .filter(ee.Filter.calendarRange(mes, mes, 'month'))
                 .filterDate(data_inicio, data_fim)
                 .filterBounds(limitesRegiao))

    # COLEÇÃO 2: ERA5-LAND (A base histórica que precisa ser calibrada)
    era5_hist = (ee.ImageCollection("ECMWF/ERA5_LAND/DAILY_AGGR")
                 .filter(ee.Filter.calendarRange(mes, mes, 'month'))
                 .filterDate(data_inicio, data_fim)
                 .filterBounds(limitesRegiao))

    def preparar_smap(img):
        # RUNOFF: Está em kg/m2/s. Multiplica-se por 86400 (segundos do dia) para virar mm/dia.
        rsup = img.select('overland_runoff_flux').multiply(86400).rename('runoff_sup')
        rsub = img.select('baseflow_flux').multiply(86400).rename('runoff_sub')

        # UMIDADE: O SMAP já entrega Fração Volumétrica (m3/m3) de fábrica. (NÃO MULTIPLICAMOS MAIS)
        swat = img.select('sm_rootzone').resample('bilinear').rename('soil_water')
        ssur = img.select('sm_surface').resample('bilinear').rename('soil_surface')

        return ee.Image.cat([rsup, rsub, swat, ssur])

    def preparar_era5(img):
        # RUNOFF: O ERA5 entrega em metros. Multiplica-se por 1000 para virar milímetros (mm).
        rsup = img.select('surface_runoff_sum').multiply(1000).max(0).rename('runoff_sup')
        rsub = img.select('sub_surface_runoff_sum').multiply(1000).max(0).rename('runoff_sub')

        # UMIDADE (0 a 5 cm): Usamos a Layer 1 do ERA5 (0-7cm) como proxy, já em Fração Volumétrica.
        ssur = img.select('volumetric_soil_water_layer_1').resample('bilinear').rename('soil_surface')

        # UMIDADE (0 a 100 cm): Média Ponderada das 3 primeiras camadas do ERA5
        # Layer 1 = 7%, Layer 2 = 21%, Layer 3 = 72% (Total = 100% de 1 metro)
        L1 = img.select('volumetric_soil_water_layer_1').multiply(0.07)
        L2 = img.select('volumetric_soil_water_layer_2').multiply(0.21)
        L3 = img.select('volumetric_soil_water_layer_3').multiply(0.72)
        swat = L1.add(L2).add(L3).resample('bilinear').rename('soil_water')

        return ee.Image.cat([rsup, rsub, swat, ssur])

    # Aplica as regras de padronização nas duas coleções
    smap_limpo = smap_hist.map(preparar_smap)
    era5_limpo = era5_hist.map(preparar_era5)

    # Calcula Média e Desvio Padrão do SMAP (Sufixo: _smap)
    img_mean_smap = smap_limpo.mean().rename(['rsup_m_smap', 'rsub_m_smap', 'swat_m_smap', 'ssur_m_smap'])
    img_std_smap = smap_limpo.reduce(ee.Reducer.stdDev()).rename(['rsup_s_smap', 'rsub_s_smap', 'swat_s_smap', 'ssur_s_smap'])

    # Calcula Média e Desvio Padrão do ERA5 (Sufixo novo: _era5)
    img_mean_era5 = era5_limpo.mean().rename(['rsup_m_era5', 'rsub_m_era5', 'swat_m_era5', 'ssur_m_era5'])
    img_std_era5 = era5_limpo.reduce(ee.Reducer.stdDev()).rename(['rsup_s_era5', 'rsub_s_era5', 'swat_s_era5', 'ssur_s_era5'])

    # Empacota as 16 matrizes estatísticas em uma única imagem final
    img_final = ee.Image.cat([img_mean_smap, img_std_smap, img_mean_era5, img_std_era5]).toFloat()

    # Rasteriza o limite territorial do GAUL para criar uma máscara de pixels eficiente
    mascara_brasil = ee.Image.constant(1).clipToCollection(fcBrasil).mask()

    # Aplica a máscara e configura o mês de referência
    return img_final.updateMask(mascara_brasil).set('mes_referencia', mes)

def expandir_borda(img):
    return img.unmask(img.reproject(crs=CRS, crsTransform=TRANSFORM).focalMean(1, 'square', 'pixels'))


def montar_correcao_mes(mes):
    """Igual a montarCorrecaoMes do painel (e à regra de aplicarPatch_Solo do motor)."""
    solo = ee.Image(f'{ASSET_SOLO}/Matriz_Solo_{mes:02d}')

    def fator(p):
        f = expandir_borda(solo.select(f'{p}_m_smap')).divide(
            expandir_borda(solo.select(f'{p}_m_era5')).max(1e-3))
        return (ee.Image.constant(1)
                .where(f.gt(FATOR_MAX).unmask(0), ee.Image.constant(FATOR_MAX).divide(f))
                .unmask(1))

    def teto(p):
        return (expandir_borda(solo.select(f'{p}_m_smap'))
                .add(expandir_borda(solo.select(f'{p}_s_smap')).multiply(SIGMAS_TETO_UMIDADE))
                .min(UMIDADE_MAX_FISICA).unmask(UMIDADE_MAX_FISICA))

    return ee.Image.cat([
        fator('rsup').rename('c_rsup'), fator('rsub').rename('c_rsub'),
        fator('ssur').rename('c_ssur'), fator('swat').rename('c_swat'),
        teto('ssur').rename('teto_ssur'), teto('swat').rename('teto_swat'),
    ])


def padronizar_nomes(img):
    """Como getSABCollection: aceita imagens com os nomes antigos de banda."""
    return ee.Image(ee.Algorithms.If(
        img.bandNames().contains('precipitation_mm'),
        img.select(NOMES_ANTIGOS, BANDAS_SAB),
        img.select(BANDAS_SAB)))


def preparar_dia(img, corr):
    """getSABCollection + corrigirSoloEscoamentoV10 + grampo termodinâmico do painel, mais as
    bandas derivadas que o painel calcula (amplitude, déficit, chuva efetiva, contagens)."""
    base = padronizar_nomes(img)
    alg = ee.String(ee.Algorithms.If(img.propertyNames().contains('versao_algoritmo'),
                                     img.get('versao_algoritmo'), 'V1.0'))
    corrigidas = ee.Image.cat([
        base.select('runoff_sup_mm').multiply(corr.select('c_rsup')).rename('runoff_sup_mm'),
        base.select('runoff_sub_mm').multiply(corr.select('c_rsub')).rename('runoff_sub_mm'),
        base.select('soil_moisture_0_5cm_m3_m3').multiply(corr.select('c_ssur'))
            .min(corr.select('teto_ssur')).rename('soil_moisture_0_5cm_m3_m3'),
        base.select('soil_moisture_0_100cm_m3_m3').multiply(corr.select('c_swat'))
            .min(corr.select('teto_swat')).rename('soil_moisture_0_100cm_m3_m3'),
    ]).toFloat()
    # Só os dias da V1.0 recebem a correção (dias V1.2, V1.3 e patch já vêm corrigidos)
    base = ee.Image(ee.Algorithms.If(alg.equals('V1.0'), base.addBands(corrigidas, None, True), base))

    et_pot = base.select('et_potencial')
    et_real = base.select('et_real').min(et_pot).rename('et_real')
    transp = base.select('transpiracao_mm').min(et_real).rename('transpiracao_mm')
    base = base.addBands(ee.Image.cat([et_real, transp]), None, True)

    tmax, tmin, pr = base.select('tmax_C'), base.select('tmin_C'), base.select('pr_mm')
    # Graus-dia (McMaster e Wilhelm, 1997, método 2): Tmax e Tmin limitadas a [TBASE, TSUP]
    gd = (tmax.max(TBASE_GD).min(TSUP_GD).add(tmin.max(TBASE_GD).min(TSUP_GD))
          .divide(2).subtract(TBASE_GD))
    derivadas = ee.Image.cat([
        tmax.add(tmin).divide(2).rename('tmed_C'),
        tmax.subtract(tmin).rename('amplitude_diaria'),
        et_pot.subtract(et_real).max(0).rename('deficit_diario'),
        pr.subtract(base.select('runoff_sup_mm')).max(0).rename('chuva_efetiva'),
        pr.subtract(et_pot).rename('balanco_hidrico'),
        gd.rename('graus_dia'),
        pr.gte(LIMIAR_DIA_SECO).rename('dias_chuva_1mm'),
        pr.lt(LIMIAR_DIA_SECO).rename('dias_secos'),
        pr.gte(LIMIAR_CHUVA_10).rename('dias_chuva_10mm'),
        pr.gte(LIMIAR_CHUVA_20).rename('dias_chuva_20mm'),
        tmin.lte(LIMIAR_FRIO).rename('dias_frios'),
        tmin.lte(LIMIAR_GEADA).rename('dias_geada'),
        tmin.lt(LIMIAR_TMIN_0).rename('dias_tmin_0'),
        tmax.gte(LIMIAR_CALOR).rename('dias_calor'),
    ]).toFloat()
    return base.addBands(derivadas).select(VARS_DIA).toFloat()

def agregar_ano_mes(ano, mes, corr):
    """Totais, médias e extremos de um mês de um ano, mascarados onde o mês não é válido."""
    ini = ee.Date.fromYMD(ano, mes, 1)
    fim = ini.advance(1, 'month')
    n_dias = ee.Image.constant(fim.difference(ini, 'day'))
    col = (ee.ImageCollection(ASSET_ORIGEM).filterDate(ini, fim)
           .map(lambda i: preparar_dia(i, corr))
           .merge(ee.ImageCollection([VAZIO_DIA])))
    totais = col.select(VARS_TOTAL).sum()
    medias = col.select(VARS_MEDIA).mean()
    extremos = ee.Image.cat([
        (col.select([orig]).max() if red == 'max' else col.select([orig]).min()).rename(nome)
        for nome, orig, red in VARS_EXTREMO])
    cont = col.select(VARS_DIA).count()
    cont_ext = ee.Image.cat([cont.select([orig]).rename(nome) for nome, orig, _ in VARS_EXTREMO])
    valido = ee.Image.cat([
        cont.select(VARS_TOTAL).eq(n_dias),                                # mês completo
        cont.select(VARS_MEDIA).gte(n_dias.subtract(FALTAS_MAX_MEDIA)),    # até 4 faltas
        cont_ext.gte(n_dias.subtract(FALTAS_MAX_MEDIA)),
    ])
    return ee.Image.cat([totais, medias, extremos]).updateMask(valido).set('ano', ano)


def total_mes_chuva_eto(ini):
    """Totais de chuva e de ETo de um mês; cada um só existe com o mês completo."""
    fim = ini.advance(1, 'month')
    n_dias = ee.Image.constant(fim.difference(ini, 'day'))
    col = (ee.ImageCollection(ASSET_ORIGEM).filterDate(ini, fim)
           .map(lambda i: padronizar_nomes(i).select(['pr_mm', 'et_potencial']))
           .merge(ee.ImageCollection([VAZIO_PR_ETO])))
    return col.sum().updateMask(col.count().eq(n_dias))


def totais_indices(ano, mes):
    """Chuva acumulada em 1, 3 e 6 meses (SPI) e chuva menos ETo em 1 e 3 meses (SPEI),
    terminando no mês dado; o acumulado só existe se todos os meses estiverem completos."""
    ini = ee.Date.fromYMD(ano, mes, 1)
    meses = [total_mes_chuva_eto(ini.advance(-k, 'month')) for k in range(max(ESCALAS_SPI + ESCALAS_SPEI))]
    pr = [t.select('pr_mm') for t in meses]
    d = [t.select('pr_mm').subtract(t.select('et_potencial')) for t in meses]

    def soma(lista):
        tot = lista[0]
        for x in lista[1:]:
            tot = tot.add(x)
        return tot

    return ee.Image.cat([soma(pr[:k]).rename(f'pr_{k}m') for k in ESCALAS_SPI] +
                        [soma(d[:k]).rename(f'd_{k}m') for k in ESCALAS_SPEI]).set('ano', ano)

def quantis_por_posto(col, banda, probs_nomes, n_min):
    """Quantis exatos pelo tipo 7 (posto (n-1)p, interpolação linear), o método que a OMM
    recomenda para quintis. Anos inválidos viram SENTINELA e ficam no fim da ordenação."""
    serie = col.select([banda])
    n = serie.count().toFloat()
    ordenado = serie.map(lambda i: i.unmask(SENTINELA)).toBands().toArray().arraySort()
    ultimo = n.subtract(1).max(0)
    saidas = []
    for p, nome in probs_nomes:
        h = n.subtract(1).multiply(p).max(0).min(ultimo)
        lo = h.floor()
        hi = lo.add(1).min(ultimo)
        frac = h.subtract(lo)
        v_lo = ordenado.arrayGet(lo.toInt())
        v_hi = ordenado.arrayGet(hi.toInt())
        saidas.append(v_lo.add(v_hi.subtract(v_lo).multiply(frac))
                      .updateMask(n.gte(n_min)).rename(nome))
    return ee.Image.cat(saidas)


def parametros_gama(col, banda, prefixo):
    """Gama para o SPI (McKee et al., 1993) pela aproximação de máxima verossimilhança de Thom:
    A = ln(média) - média(ln), alfa = (1 + raiz(1 + 4A/3)) / 4A, beta = média / alfa,
    ajustada só aos anos com chuva; q0 = fração de anos sem chuva."""
    serie = col.select([banda])
    n = serie.count().toFloat()
    positivos = serie.map(lambda i: i.updateMask(i.gte(LIMIAR_ZERO_CHUVA_MM)))
    n_pos = positivos.count().toFloat()
    media_pos = positivos.mean()
    media_ln = positivos.map(lambda i: i.log()).mean()
    A = media_pos.log().subtract(media_ln)
    alfa = A.multiply(4.0 / 3.0).add(1).sqrt().add(1).divide(A.multiply(4))
    beta = media_pos.divide(alfa)
    q0 = n.subtract(n_pos).divide(n)
    valido = n.gte(ANOS_MINIMOS).And(n_pos.gte(MIN_ANOS_COM_CHUVA_SPI)).And(A.gt(1e-6))
    return ee.Image.cat([
        alfa.updateMask(valido).rename(f'{prefixo}_alfa'),
        beta.updateMask(valido).rename(f'{prefixo}_beta'),
        q0.updateMask(valido).rename(f'{prefixo}_q0'),
        n.rename(f'{prefixo}_n'),
    ])


def parametros_glo(col, banda, prefixo, n_anos):
    """SPEI: log-logística de 3 parâmetros (Vicente-Serrano et al., 2010) escrita como logística
    generalizada de Hosking, a forma do pacote SPEI do R (lmom::pelglo e cdfglo). Momentos-L
    amostrais pelos estimadores não viciados a_s = (1/n) soma C(n-i, s)/C(n-1, s) x_(i):
    l1 = a0, l2 = a0 - 2a1, l3 = a0 - 6a1 + 6a2. Parâmetros (Hosking e Wallis, 1997):
    kappa = -l3/l2, alfa = l2 sen(kappa pi)/(kappa pi), xi = l1 - alfa (1/kappa - pi/sen(kappa pi)).
    Com assimetria positiva é a mesma curva de Vicente-Serrano (beta = -1/kappa); com assimetria
    negativa ou nula, onde a forma original não tem solução, a curva continua definida.
    Cada valor ordenado vira uma banda e os pesos dependem só de n (anos válidos do pixel)."""
    serie = col.select([banda])
    n = serie.count().toDouble()
    nomes = [f'v{i:02d}' for i in range(1, n_anos + 1)]
    ordenado = (serie.map(lambda i: i.toDouble().unmask(SENTINELA)).toBands().toArray().arraySort()
                .arrayFlatten([nomes]))
    a0 = serie.map(lambda i: i.toDouble()).mean()
    a1 = ee.Image.constant(0).toDouble()
    a2 = ee.Image.constant(0).toDouble()
    for i in range(1, n_anos + 1):
        x = ordenado.select(nomes[i - 1])
        dentro = n.gte(i)                                  # posições além de n são SENTINELA: peso 0
        c1 = n.subtract(i).divide(n.subtract(1)).multiply(dentro)
        c2 = (n.subtract(i).multiply(n.subtract(i + 1))
              .divide(n.subtract(1).multiply(n.subtract(2))).multiply(dentro))
        a1 = a1.add(x.multiply(c1))
        a2 = a2.add(x.multiply(c2))
    a1 = a1.divide(n)
    a2 = a2.divide(n)
    l1 = a0
    l2 = a0.subtract(a1.multiply(2))
    l3 = a0.subtract(a1.multiply(6)).add(a2.multiply(6))
    kappa = l3.divide(l2).multiply(-1)
    kappa = kappa.where(kappa.abs().lt(KAPPA_MINIMO), KAPPA_MINIMO)
    kpi = kappa.multiply(math.pi)
    alfa = l2.multiply(kpi.sin()).divide(kpi)
    xi = l1.subtract(alfa.multiply(ee.Image.constant(1).divide(kappa).subtract(
        ee.Image.constant(math.pi).divide(kpi.sin()))))
    valido = n.gte(ANOS_MINIMOS).And(l2.gt(0)).And(kappa.abs().lt(1)).And(alfa.gt(0))
    return ee.Image.cat([
        xi.updateMask(valido).rename(f'{prefixo}_xi'),
        alfa.updateMask(valido).rename(f'{prefixo}_alfa'),
        kappa.updateMask(valido).rename(f'{prefixo}_kappa'),
        n.rename(f'{prefixo}_n'),
    ])


def percentis_chuva_umida(p, mes):
    """P95 e P99 da chuva diária nos dias com chuva (>= 1 mm) do mês, nos 30 anos juntos
    (limiar de chuva intensa do mês, no espírito do R95p do ETCCDI)."""
    a0, a1 = PERIODOS[p]['anos']
    col = (ee.ImageCollection(ASSET_ORIGEM).filter(ee.Filter.calendarRange(mes, mes, 'month'))
           .filterDate(f'{a0}-01-01', f'{a1 + 1}-01-01')
           .map(lambda i: padronizar_nomes(i).select(['pr_mm']))
           .map(lambda i: i.updateMask(i.gte(LIMIAR_DIA_SECO))))
    n = col.count()
    pct = (col.reduce(ee.Reducer.percentile(percentiles=[95, 99], maxRaw=1000))
           .rename(['pr_p95_umido', 'pr_p99_umido']).updateMask(n.gte(MIN_DIAS_UMIDOS)))
    return ee.Image.cat([pct, n.rename('n_dias_umidos')])


def nomes_bandas_normais():
    return ([f'{v}_media' for v in VARS_MES] + [f'{v}_dp' for v in VARS_MES] +
            [f'{v}_n' for v in VARS_MES] + [nome for _, nome in QUANTIS_CHUVA] +
            [f'{v}_{s}' for v in ('tmax_C', 'tmin_C') for _, s in TERCIS_TEMP] +
            [f'spi{k}_{c}' for k in ESCALAS_SPI for c in ('alfa', 'beta', 'q0', 'n')] +
            [f'spei{k}_{c}' for k in ESCALAS_SPEI for c in ('xi', 'alfa', 'kappa', 'n')] +
            ['pr_p95_umido', 'pr_p99_umido', 'n_dias_umidos'])


def imagem_normais(p, mes):
    a0, a1 = PERIODOS[p]['anos']
    corr = montar_correcao_mes(mes)
    anos = ee.List.sequence(a0, a1)
    anual = ee.ImageCollection.fromImages(anos.map(lambda a: agregar_ano_mes(ee.Number(a), mes, corr)))
    n = anual.count()                                   # 33 bandas, na ordem de VARS_MES
    ok = n.gte(ANOS_MINIMOS)
    media = anual.mean().updateMask(ok).rename([f'{v}_media' for v in VARS_MES])
    dp = anual.reduce(ee.Reducer.sampleStdDev()).updateMask(ok).rename([f'{v}_dp' for v in VARS_MES])
    contagem = n.rename([f'{v}_n' for v in VARS_MES])
    quantis = quantis_por_posto(anual, 'pr_mm', QUANTIS_CHUVA, ANOS_MINIMOS)
    tercis = ee.Image.cat([
        quantis_por_posto(anual, v, [(q, f'{v}_{s}') for q, s in TERCIS_TEMP], ANOS_MINIMOS)
        for v in ('tmax_C', 'tmin_C')])
    acumulados = ee.ImageCollection.fromImages(anos.map(lambda a: totais_indices(ee.Number(a), mes)))
    spi = ee.Image.cat([parametros_gama(acumulados, f'pr_{k}m', f'spi{k}') for k in ESCALAS_SPI])
    spei = ee.Image.cat([parametros_glo(acumulados, f'd_{k}m', f'spei{k}', a1 - a0 + 1)
                         for k in ESCALAS_SPEI])
    # select com a lista fixa: ordem garantida e erro imediato se faltar alguma banda que o painel lê
    img = ee.Image.cat([media, dp, contagem, quantis, tercis, spi, spei,
                        percentis_chuva_umida(p, mes)]).select(nomes_bandas_normais()).toFloat()
    return img.set({
        'mes': mes, 'periodo': p, 'ano_inicio': a0, 'ano_fim': a1,
        'dias_medios_mes': dias_medios_mes(a0, a1, mes),
        'anos_minimos': ANOS_MINIMOS, 'faltas_max_media': FALTAS_MAX_MEDIA,
        'limiar_zero_chuva_mm': LIMIAR_ZERO_CHUVA_MM, 'min_anos_com_chuva_spi': MIN_ANOS_COM_CHUVA_SPI,
        'limiar_dia_seco_mm': LIMIAR_DIA_SECO, 'limiar_chuva_10_mm': LIMIAR_CHUVA_10,
        'limiar_chuva_20_mm': LIMIAR_CHUVA_20, 'limiar_frio_C': LIMIAR_FRIO, 'limiar_geada_C': LIMIAR_GEADA,
        'limiar_tmin_0_C': LIMIAR_TMIN_0, 'limiar_calor_C': LIMIAR_CALOR,
        'tbase_graus_dia_C': TBASE_GD, 'tsup_graus_dia_C': TSUP_GD, 'min_dias_umidos': MIN_DIAS_UMIDOS,
        'kappa_minimo_spei': KAPPA_MINIMO,
        'versao_baseline': VERSAO_BASELINE, 'origem': ASSET_ORIGEM, 'gerado_em': agora_brt,
        'metodo': ('OMM WMO-No.1203: total so com mes completo; media e extremos com ate 4 faltas; normal '
                   'com >=24 anos; dp amostral entre anos; quantis tipo 7; SPI gama (Thom); SPEI '
                   'logistica generalizada de Hosking (momentos-L nao viciados, pelglo/cdfglo do pacote '
                   'SPEI); leitura igual ao Painel SAB (correcao V1.0 de solo/escoamento e ET real <= ETo)'),
    })


def imagem_limiares(p, mes, dias=None):
    """P90 e P10 da Tmax e da Tmin de cada dia do calendário, com janela de 5 dias (±2) em todos
    os anos da normal (ETCCDI: TX90p, TX10p, TN90p, TN10p). Dias da janela fora da normal ficam
    de fora."""
    a0, a1 = PERIODOS[p]['anos']
    lim_ini, lim_fim = date(a0, 1, 1), date(a1, 12, 31)
    base = ee.ImageCollection(ASSET_ORIGEM).filterDate(f'{a0}-01-01', f'{a1 + 1}-01-01')
    n_dias_mes = 28 if mes == 2 else calendar.monthrange(2001, mes)[1]
    dias = list(dias) if dias else list(range(1, n_dias_mes + 1))
    n_min = math.ceil(FRACAO_MIN_JANELA * (2 * JANELA_LIMIARES + 1) * (a1 - a0 + 1))
    reducer = ee.Reducer.percentile(percentiles=[10, 90], maxRaw=1000)   # cálculo direto (<= 150 valores)
    grupos = {'tx90': [], 'tx10': [], 'tn90': [], 'tn10': []}
    for d in dias:
        filtros = []
        for a in range(a0, a1 + 1):
            centro = date(a, mes, d)
            ini = max(centro - timedelta(days=JANELA_LIMIARES), lim_ini)
            fim = min(centro + timedelta(days=JANELA_LIMIARES), lim_fim)
            filtros.append(ee.Filter.date(ini.isoformat(), (fim + timedelta(days=1)).isoformat()))
        janela = base.filter(ee.Filter.Or(*filtros)).map(
            lambda i: padronizar_nomes(i).select(['tmax_C', 'tmin_C']))
        for var, pref in (('tmax_C', 'tx'), ('tmin_C', 'tn')):
            serie = janela.select([var])
            ok = serie.count().gte(n_min)
            pct = serie.reduce(reducer).rename([f'{pref}10', f'{pref}90']).updateMask(ok)
            grupos[f'{pref}90'].append(pct.select(f'{pref}90').rename(f'{pref}90_d{d:02d}'))
            grupos[f'{pref}10'].append(pct.select(f'{pref}10').rename(f'{pref}10_d{d:02d}'))
    return ee.Image.cat(grupos['tx90'] + grupos['tx10'] + grupos['tn90'] + grupos['tn10']).toFloat().set({
        'mes': mes, 'periodo': p, 'ano_inicio': a0, 'ano_fim': a1, 'dias_no_mes': n_dias_mes,
        'janela_dias': 2 * JANELA_LIMIARES + 1, 'percentis': '10 e 90', 'minimo_dias_janela': n_min,
        'versao_baseline': VERSAO_BASELINE, 'origem': ASSET_ORIGEM, 'gerado_em': agora_brt,
        'metodo': 'ETCCDI: percentil do dia do calendario com janela de 5 dias, sem bootstrap',
    })

def maior_sequencia(dias_faltando):
    maior, atual, anterior = 0, 0, None
    for d in sorted(dias_faltando):
        atual = atual + 1 if anterior is not None and (d - anterior).days == 1 else 1
        maior = max(maior, atual)
        anterior = d
    return maior


def inventario(p):
    a0, a1 = PERIODOS[p]['anos']
    col = ee.ImageCollection(ASSET_ORIGEM).filterDate(f'{a0}-01-01', f'{a1 + 1}-01-01')

    def para_feature(img):
        d = img.toDictionary()
        return ee.Feature(None, {
            't': img.get('system:time_start'),
            'versao': d.get('versao', 'definitiva'),
            'alg': d.get('versao_algoritmo', 'V1.0'),
            'antigo': img.bandNames().contains('precipitation_mm'),
        })

    t, versao, alg, antigo = col.map(para_feature).reduceColumns(
        ee.Reducer.toList().repeat(4), ['t', 'versao', 'alg', 'antigo']).get('list').getInfo()
    datas = pd.to_datetime(pd.Series(t, dtype='float64'), unit='ms', utc=True).dt.tz_localize(None).dt.normalize()
    esperadas = pd.date_range(f'{a0}-01-01', f'{a1}-12-31')
    presentes = set(datas)
    faltando = sorted(set(esperadas) - presentes)
    duplicadas = sorted(set(datas[datas.duplicated()]))

    linhas = []
    for a in range(a0, a1 + 1):
        for m in range(1, 13):
            f = [d for d in faltando if d.year == a and d.month == m]
            linhas.append({'ano': a, 'mes': m, 'faltas': len(f), 'seguidas': maior_sequencia(f),
                           'duplicadas': sum(1 for d in duplicadas if d.year == a and d.month == m)})
    meses = pd.DataFrame(linhas)
    meses['total_ok'] = (meses.faltas == 0) & (meses.duplicadas == 0)
    meses['media_ok_omm'] = (meses.faltas <= 10) & (meses.seguidas <= 4)
    meses['media_ok_script'] = meses.faltas <= FALTAS_MAX_MEDIA
    anos_validos = meses.groupby('mes')[['total_ok', 'media_ok_omm', 'media_ok_script']].sum()

    print(f"\n📋 INVENTÁRIO {p.replace('_', '-')}: {len(datas)} imagens | esperadas {len(esperadas)} | "
          f"faltando {len(faltando)} | datas duplicadas {len(duplicadas)}")
    print(f"   versão: {pd.Series(versao).value_counts().to_dict()} | algoritmo: "
          f"{pd.Series(alg).value_counts().to_dict()} | com nomes antigos de banda: {int(sum(bool(x) for x in antigo))}")
    if faltando:
        print(f"   primeiros dias faltando: {[d.strftime('%Y-%m-%d') for d in faltando[:10]]}")
    if duplicadas:
        print(f"   ⚠️ datas duplicadas (somariam duas vezes): {[d.strftime('%Y-%m-%d') for d in duplicadas[:10]]}")
    if (anos_validos < 30).any().any():
        print("   Anos válidos por mês (de 30):")
        print(anos_validos.to_string())
    DATAS_NAO_V10[p] = {d for d, a in zip(datas, alg) if a != 'V1.0'}
    nao_def = sum(1 for v in versao if v != 'definitiva')
    if nao_def:
        print(f"   ⚠️ {nao_def} dia(s) não definitivos dentro da normal")
    min_anos = int(anos_validos.min().min())
    # Faltas isoladas são tratadas pela regra da OMM; duplicatas indicam banco inconsistente
    ok = (not duplicadas and nao_def == 0 and min_anos >= ANOS_MINIMOS)
    return {'ok': ok, 'faltando': len(faltando), 'duplicadas': len(duplicadas),
            'min_anos': min_anos, 'nao_definitivos': nao_def}

def centro_pixel(lat, lon):
    dx, x0 = TRANSFORM[0], TRANSFORM[2]
    dy, y0 = -TRANSFORM[4], TRANSFORM[5]
    return (y0 - (math.floor((y0 - lat) / dy) + 0.5) * dy,
            x0 + (math.floor((lon - x0) / dx) + 0.5) * dx)


def pontos_conferencia():
    return [{'nome': n, 'lat': centro_pixel(la, lo)[0], 'lon': centro_pixel(la, lo)[1]}
            for n, la, lo in PONTOS_REFERENCIA]


def amostrar(img, pontos):
    """Valor de cada banda no pixel de cada ponto; se o Earth Engine recusar o pedido conjunto
    (tempo ou memória), pede um ponto por vez."""
    def ler(lista):
        fc = ee.FeatureCollection([ee.Feature(ee.Geometry.Point([q['lon'], q['lat']]), {'nome': q['nome']})
                                   for q in lista])
        feats = img.reduceRegions(collection=fc, reducer=ee.Reducer.first(), crs=CRS,
                                  crsTransform=TRANSFORM).getInfo()['features']
        return {f['properties']['nome']: f['properties'] for f in feats}
    try:
        return ler(pontos)
    except Exception as e:
        print(f"   ↪️ pedido conjunto recusado ({str(e)[:80]}); um ponto por vez")
        res = {}
        for q in pontos:
            res.update(ler([q]))
        return res


ERROS_DE_LIMITE = ('memory', 'timed out', 'Too many', 'too large', 'exceeded', 'Computation')


def serie_pontos(col, bandas, pontos, ini, fim):
    """Série diária de cada ponto (getRegion na grade do banco) entre as datas ini e fim.
    Se o Earth Engine recusar o pedido (memória ou tempo), divide o período ao meio até blocos
    de cerca de um ano e, se ainda for preciso, lê um ponto por vez."""
    avisado = []

    def ler(lista, a, b):
        geom = ee.Geometry.MultiPoint([[q['lon'], q['lat']] for q in lista])
        linhas = (col.filterDate(a.isoformat(), b.isoformat()).select(bandas)
                  .getRegion(geom, None, CRS, TRANSFORM).getInfo())
        return pd.DataFrame(linhas[1:], columns=linhas[0])

    def ler_adaptativo(lista, a, b):
        try:
            return ler(lista, a, b)
        except ee.EEException as e:
            if not any(m in str(e) for m in ERROS_DE_LIMITE):
                raise
            if not avisado:
                print(f"   ↪️ leitura recusada pelo Earth Engine ({str(e)[:60]}); lendo em blocos menores")
                avisado.append(1)
            dias = (b - a).days
            if dias > 400:
                meio = a + timedelta(days=dias // 2)
                return pd.concat([ler_adaptativo(lista, a, meio), ler_adaptativo(lista, meio, b)], ignore_index=True)
            if len(lista) > 1:
                return pd.concat([ler_adaptativo([q], a, b) for q in lista], ignore_index=True)
            raise

    df = ler_adaptativo(list(pontos), ini, fim)
    lat = df['latitude'].astype(float).to_numpy()
    lon = df['longitude'].astype(float).to_numpy()
    dist = np.stack([np.hypot(lat - q['lat'], lon - q['lon']) for q in pontos])
    df['nome'] = [pontos[i]['nome'] for i in dist.argmin(axis=0)] if len(df) else []
    df['data'] = pd.to_datetime(df['time'].astype('float64'), unit='ms', utc=True).dt.tz_localize(None).dt.normalize()
    for b in bandas:
        df[b] = pd.to_numeric(df[b], errors='coerce')
    return df[['nome', 'data'] + bandas].sort_values(['nome', 'data'], ignore_index=True)


def preparar_local(bruta, corr_pts, datas_nao_v10):
    """A leitura de cada dia do painel (preparar_dia) refeita em pandas a partir das bandas
    brutas: correção V1.0 de escoamento e umidade do solo com os fatores do mês amostrados no
    pixel, ET real limitada à ETo, transpiração limitada à ET real e as bandas derivadas.
    Valor ausente continua ausente, como a máscara do Earth Engine."""
    s = bruta.copy()
    v10 = ~s['data'].isin(datas_nao_v10)
    def campo(c):
        return s['nome'].map(lambda n: float(corr_pts[n][c]))
    for banda, fator, teto in (('runoff_sup_mm', 'c_rsup', None), ('runoff_sub_mm', 'c_rsub', None),
                               ('soil_moisture_0_5cm_m3_m3', 'c_ssur', 'teto_ssur'),
                               ('soil_moisture_0_100cm_m3_m3', 'c_swat', 'teto_swat')):
        novo = s[banda] * campo(fator)
        if teto:
            novo = np.minimum(novo, campo(teto))
        s[banda] = novo.where(v10, s[banda])
    s['et_real'] = np.minimum(s['et_real'], s['et_potencial'])
    s['transpiracao_mm'] = np.minimum(s['transpiracao_mm'], s['et_real'])
    pr, tx, tn, eto = s['pr_mm'], s['tmax_C'], s['tmin_C'], s['et_potencial']
    s['tmed_C'] = (tx + tn) / 2
    s['amplitude_diaria'] = tx - tn
    s['deficit_diario'] = np.maximum(eto - s['et_real'], 0)
    s['chuva_efetiva'] = np.maximum(pr - s['runoff_sup_mm'], 0)
    s['balanco_hidrico'] = pr - eto
    s['graus_dia'] = (tx.clip(TBASE_GD, TSUP_GD) + tn.clip(TBASE_GD, TSUP_GD)) / 2 - TBASE_GD
    def marca(base, cond):
        return cond.astype(float).where(base.notna())
    s['dias_chuva_1mm'] = marca(pr, pr >= LIMIAR_DIA_SECO)
    s['dias_secos'] = marca(pr, pr < LIMIAR_DIA_SECO)
    s['dias_chuva_10mm'] = marca(pr, pr >= LIMIAR_CHUVA_10)
    s['dias_chuva_20mm'] = marca(pr, pr >= LIMIAR_CHUVA_20)
    s['dias_frios'] = marca(tn, tn <= LIMIAR_FRIO)
    s['dias_geada'] = marca(tn, tn <= LIMIAR_GEADA)
    s['dias_tmin_0'] = marca(tn, tn < LIMIAR_TMIN_0)
    s['dias_calor'] = marca(tx, tx >= LIMIAR_CALOR)
    return s[['nome', 'data'] + VARS_DIA]


def quantil_local(valores, p, tipo):
    x = np.sort(np.asarray(valores, dtype=float))
    n = len(x)
    if n == 0:
        return np.nan
    h = (n - 1) * p if tipo == 7 else (n + 1 / 3) * p + 1 / 3 - 1
    h = min(max(h, 0.0), n - 1.0)
    lo = int(math.floor(h))
    hi = min(lo + 1, n - 1)
    return x[lo] + (h - lo) * (x[hi] - x[lo])


def gama_local(totais):
    v = np.asarray([t for t in totais if np.isfinite(t)], dtype=float)
    n = len(v)
    pos = v[v >= LIMIAR_ZERO_CHUVA_MM]
    res = {'alfa': np.nan, 'beta': np.nan, 'q0': np.nan, 'n': float(n)}
    if n < ANOS_MINIMOS or len(pos) < MIN_ANOS_COM_CHUVA_SPI:
        return res
    A = math.log(pos.mean()) - np.log(pos).mean()
    if A <= 1e-6:
        return res
    alfa = (1 + math.sqrt(1 + 4 * A / 3)) / (4 * A)
    res.update(alfa=alfa, beta=pos.mean() / alfa, q0=(n - len(pos)) / n)
    return res


def glo_local(valores):
    """Mesmo ajuste do Earth Engine (momentos-L não viciados e logística generalizada), feito
    aqui de forma independente."""
    v = np.sort(np.asarray([t for t in valores if np.isfinite(t)], dtype=float))
    n = len(v)
    res = {'xi': np.nan, 'alfa': np.nan, 'kappa': np.nan, 'n': float(n)}
    if n < ANOS_MINIMOS:
        return res
    i = np.arange(1, n + 1)
    a0 = v.mean()
    a1 = np.sum((n - i) / (n - 1) * v) / n
    a2 = np.sum((n - i) * (n - i - 1) / ((n - 1) * (n - 2)) * v) / n
    l1, l2, l3 = a0, a0 - 2 * a1, a0 - 6 * a1 + 6 * a2
    if not l2 > 0:
        return res
    k = -l3 / l2
    if abs(k) < KAPPA_MINIMO:
        k = KAPPA_MINIMO
    if not abs(k) < 1:
        return res
    alfa = l2 * math.sin(k * math.pi) / (k * math.pi)
    if not alfa > 0:
        return res
    res.update(xi=l1 - alfa * (1 / k - math.pi / math.sin(k * math.pi)), alfa=alfa, kappa=k)
    return res


def valores_mensais_locais(serie, a0, a1, mes):
    """Totais, médias e extremos de cada ano pela mesma regra do Earth Engine (NaN = mês inválido)."""
    linhas = []
    for a in range(a0, a1 + 1):
        n_dias = calendar.monthrange(a, mes)[1]
        g = serie[(serie.data.dt.year == a) & (serie.data.dt.month == mes)]
        lin = {'ano': a}
        for v in VARS_DIA:
            s = g[v].dropna()
            if v in VARS_TOTAL:
                lin[v] = s.sum() if len(s) == n_dias else np.nan
            else:
                lin[v] = s.mean() if len(s) >= n_dias - FALTAS_MAX_MEDIA else np.nan
        for nome_e, orig, red in VARS_EXTREMO:
            s = g[orig].dropna()
            ok = len(s) > 0 and len(s) >= n_dias - FALTAS_MAX_MEDIA
            lin[nome_e] = (s.max() if red == 'max' else s.min()) if ok else np.nan
        linhas.append(lin)
    return pd.DataFrame(linhas)


def totais_indices_locais(serie, a0, a1, mes):
    """Acumulados de chuva (SPI) e de chuva menos ETo (SPEI) terminados no mês, por ano."""
    def total(a, m, banda):
        while m < 1:
            a, m = a - 1, m + 12
        n_dias = calendar.monthrange(a, m)[1]
        s = serie[(serie.data.dt.year == a) & (serie.data.dt.month == m)][banda].dropna()
        return s.sum() if len(s) == n_dias else np.nan
    res = {f'pr_{k}m': [] for k in ESCALAS_SPI}
    res.update({f'd_{k}m': [] for k in ESCALAS_SPEI})
    kmax = max(ESCALAS_SPI + ESCALAS_SPEI)
    for a in range(a0, a1 + 1):
        pr = [total(a, mes - k, 'pr_mm') for k in range(kmax)]
        et = [total(a, mes - k, 'et_potencial') for k in range(kmax)]
        for k in ESCALAS_SPI:
            res[f'pr_{k}m'].append(sum(pr[:k]))
        for k in ESCALAS_SPEI:
            res[f'd_{k}m'].append(sum(x - y for x, y in zip(pr[:k], et[:k])))
    return res


def normais_locais(serie_prep, serie_base, a0, a1, mes):
    res = {}
    for nome in serie_prep.nome.unique():
        anual = valores_mensais_locais(serie_prep[serie_prep.nome == nome], a0, a1, mes)
        r = {}
        for v in VARS_MES:
            x = anual[v].dropna().to_numpy()
            n = len(x)
            r[f'{v}_n'] = float(n)
            r[f'{v}_media'] = x.mean() if n >= ANOS_MINIMOS else np.nan
            r[f'{v}_dp'] = x.std(ddof=1) if n >= ANOS_MINIMOS else np.nan
        pr = anual['pr_mm'].dropna().to_numpy()
        for p, nome_q in QUANTIS_CHUVA:
            r[nome_q] = quantil_local(pr, p, 7) if len(pr) >= ANOS_MINIMOS else np.nan
        for v in ('tmax_C', 'tmin_C'):
            x = anual[v].dropna().to_numpy()
            for q, s in TERCIS_TEMP:
                r[f'{v}_{s}'] = quantil_local(x, q, 7) if len(x) >= ANOS_MINIMOS else np.nan
        acum = totais_indices_locais(serie_base[serie_base.nome == nome], a0, a1, mes)
        for k in ESCALAS_SPI:
            g = gama_local(acum[f'pr_{k}m'])
            for c in ('alfa', 'beta', 'q0', 'n'):
                r[f'spi{k}_{c}'] = g[c]
        for k in ESCALAS_SPEI:
            g = glo_local(acum[f'd_{k}m'])
            for c in ('xi', 'alfa', 'kappa', 'n'):
                r[f'spei{k}_{c}'] = g[c]
        res[nome] = r
    return res


def chuva_umida_local(serie_base, a0, a1, mes):
    """Chuva dos dias com chuva do mês nos anos da normal (para conferir P95 e P99)."""
    res = {}
    for nome in serie_base.nome.unique():
        s = serie_base[(serie_base.nome == nome) & (serie_base.data.dt.month == mes) &
                       (serie_base.data.dt.year >= a0) & (serie_base.data.dt.year <= a1)]
        x = np.sort(s['pr_mm'].dropna().to_numpy())
        x = x[x >= LIMIAR_DIA_SECO]
        res[nome] = {'pr_p95_umido': x, 'pr_p99_umido': x}

def limiares_locais(serie_temp, a0, a1, mes, dias):
    """Valores da janela de cada dia, para conferir os percentis do Earth Engine."""
    lim_ini, lim_fim = pd.Timestamp(a0, 1, 1), pd.Timestamp(a1, 12, 31)
    res = {}
    for nome in serie_temp.nome.unique():
        s = serie_temp[serie_temp.nome == nome]
        r = {}
        for d in dias:
            sel = np.zeros(len(s), dtype=bool)
            for a in range(a0, a1 + 1):
                c = pd.Timestamp(a, mes, d)
                ini = max(c - pd.Timedelta(days=JANELA_LIMIARES), lim_ini)
                fim = min(c + pd.Timedelta(days=JANELA_LIMIARES), lim_fim)
                sel |= ((s.data >= ini) & (s.data <= fim)).to_numpy()
            tx = np.sort(s.loc[sel, 'tmax_C'].dropna().to_numpy())
            tn = np.sort(s.loc[sel, 'tmin_C'].dropna().to_numpy())
            r[f'tx90_d{d:02d}'], r[f'tx10_d{d:02d}'] = tx, tx
            r[f'tn90_d{d:02d}'], r[f'tn10_d{d:02d}'] = tn, tn
        res[nome] = r
    return res


def comparar(ee_vals, loc_vals, rtol=5e-4, atol=1e-4):
    linhas = []
    for nome, loc in loc_vals.items():
        e = ee_vals.get(nome, {})
        for banda, ref in loc.items():
            v = e.get(banda)
            v = np.nan if v is None else float(v)
            ambos_nan = not np.isfinite(v) and not np.isfinite(ref)
            ok = ambos_nan or (np.isfinite(v) and np.isfinite(ref) and abs(v - ref) <= atol + rtol * abs(ref))
            linhas.append({'ponto': nome, 'banda': banda, 'ee': v, 'local': ref, 'ok': ok,
                           'dif': abs(v - ref) if np.isfinite(v) and np.isfinite(ref) else np.nan})
    return pd.DataFrame(linhas)


def percentil_da_banda(banda):
    for marca, p in (('p95', 0.95), ('p99', 0.99), ('90_', 0.90), ('10_', 0.10)):
        if marca in banda:
            return p
    raise ValueError(banda)


def comparar_percentis(ee_vals, amostras, n_min):
    """O percentil do Earth Engine (cálculo direto sobre os valores) tem de cair entre as
    estatísticas de ordem vizinhas da posição do percentil; também diz qual definição bate."""
    linhas = []
    for nome, jan in amostras.items():
        for banda, x in jan.items():
            p = percentil_da_banda(banda)
            v = ee_vals.get(nome, {}).get(banda)
            v = np.nan if v is None else float(v)
            n = len(x)
            if n < n_min:
                linhas.append({'ponto': nome, 'banda': banda, 'n': n, 'ee': v, 'ok': not np.isfinite(v),
                               'tipo7': np.nan, 'tipo8': np.nan, 'faixa': ''})
                continue
            h8 = (n + 1 / 3) * p + 1 / 3 - 1
            lo, hi = max(int(math.floor(h8)) - 1, 0), min(int(math.ceil(h8)) + 1, n - 1)
            t7, t8 = quantil_local(x, p, 7), quantil_local(x, p, 8)
            ok = np.isfinite(v) and x[lo] - 1e-4 <= v <= x[hi] + 1e-4
            linhas.append({'ponto': nome, 'banda': banda, 'n': n, 'ee': v, 'ok': ok,
                           'tipo7': t7, 'tipo8': t8, 'faixa': f'{x[lo]:.2f} a {x[hi]:.2f}'})
    return pd.DataFrame(linhas)


def grupo_banda(b):
    if b.startswith('spei'):
        return 'SPEI ' + b[4]
    if b.startswith('spi'):
        return 'SPI ' + b[3]
    if b.startswith('pr_mm_q') or b.startswith('pr_mm_t') or b == 'pr_mm_mediana':
        return 'quantis da chuva'
    if b.endswith('_t33') or b.endswith('_t67'):
        return 'tercis de temperatura'
    if b == 'n_dias_umidos':
        return 'dias úmidos (P95)'
    for suf, rot in (('_media', 'médias'), ('_dp', 'desvios'), ('_n', 'anos válidos')):
        if b.endswith(suf):
            return rot
    return 'outras'


def verificar(p, mes):
    a0, a1 = PERIODOS[p]['anos']
    pontos = pontos_conferencia()
    t0 = time.time()
    print(f"\n🔬 CONFERÊNCIA {p.replace('_', '-')}, mês {mes:02d}, em {len(pontos)} pontos")

    print("   ⏳ normais e limiares calculados pelo Earth Engine...")
    ee_n = amostrar(imagem_normais(p, mes), pontos)
    ee_l = amostrar(imagem_limiares(p, mes, DIAS_VERIFICACAO), pontos)

    print("   ⏳ séries diárias brutas dos mesmos pixels (a leitura do painel é refeita aqui)...")
    corr_pts = amostrar(montar_correcao_mes(mes), pontos)
    if p not in DATAS_NAO_V10:
        inventario(p)
    col_mes = (ee.ImageCollection(ASSET_ORIGEM).filter(ee.Filter.calendarRange(mes, mes, 'month'))
               .map(padronizar_nomes))
    serie_bruta = serie_pontos(col_mes, BANDAS_SAB, pontos, date(a0, 1, 1), date(a1 + 1, 1, 1))
    serie_prep = preparar_local(serie_bruta, corr_pts, DATAS_NAO_V10[p])
    kmax = max(ESCALAS_SPI + ESCALAS_SPEI)
    meses_base = {((mes - k - 1) % 12) + 1 for k in range(kmax)}
    if any(d + JANELA_LIMIARES > 28 for d in DIAS_VERIFICACAO):      # janela entra no mês seguinte
        meses_base.add(mes % 12 + 1)
    filtro_meses = ee.Filter.Or(*[ee.Filter.calendarRange(m, m, 'month') for m in sorted(meses_base)])
    bandas_base = ['pr_mm', 'et_potencial', 'tmax_C', 'tmin_C']
    col_base = (ee.ImageCollection(ASSET_ORIGEM).filter(filtro_meses)
                .map(lambda i: padronizar_nomes(i).select(bandas_base)))
    serie_base = serie_pontos(col_base, bandas_base, pontos, date(a0 - 1, 7, 1), date(a1 + 1, 1, 1))

    loc_n = normais_locais(serie_prep, serie_base, a0, a1, mes)
    umidos = chuva_umida_local(serie_base, a0, a1, mes)
    for nome, r in loc_n.items():
        r['n_dias_umidos'] = float(len(umidos.get(nome, {}).get('pr_p95_umido', [])))
    janelas = limiares_locais(serie_base, a0, a1, mes, DIAS_VERIFICACAO)
    n_min = math.ceil(FRACAO_MIN_JANELA * (2 * JANELA_LIMIARES + 1) * (a1 - a0 + 1))

    comp = comparar(ee_n, loc_n)
    comp['grupo'] = comp.banda.map(grupo_banda)
    resumo = comp.groupby('grupo').agg(comparacoes=('ok', 'size'), iguais=('ok', 'sum'),
                                       maior_diferenca=('dif', 'max'))
    comp_l = pd.concat([comparar_percentis(ee_l, janelas, n_min),
                        comparar_percentis(ee_n, umidos, MIN_DIAS_UMIDOS)], ignore_index=True)
    print(f"\n   EARTH ENGINE x CÁLCULO LOCAL (tolerância relativa 5e-4) | {time.time() - t0:.0f} s")
    print(resumo.to_string())
    print(f"   percentis diários (limiares dos dias {DIAS_VERIFICACAO} e P95/P99 da chuva): "
          f"{int(comp_l.ok.sum())} de {len(comp_l)} dentro da faixa das estatísticas de ordem vizinhas")
    temp = comp_l[comp_l.banda.str.startswith('t') & comp_l.ok]
    if len(temp):
        d7 = (temp.ee - temp.tipo7).abs().max()
        d8 = (temp.ee - temp.tipo8).abs().max()
        print(f"   percentil do Earth Engine nos limiares: maior diferença {d7:.3f} °C para o tipo 7 "
              f"e {d8:.3f} °C para o tipo 8")
    erros = comp[~comp.ok]
    if len(erros):
        print("   ❌ divergências (até 15):")
        print(erros.head(15).to_string(index=False))
    if (~comp_l.ok).any():
        print(comp_l[~comp_l.ok].to_string(index=False))

    # Coerência física e da regra dos 80%
    dm = dias_medios_mes(a0, a1, mes)
    problemas = []
    for nome, r in loc_n.items():
        e = {k: (np.nan if v is None else v) for k, v in ee_n.get(nome, {}).items() if k != 'nome'}
        g = lambda k: float(e.get(k, np.nan))  # noqa: E731
        if min(g(f'{v}_n') for v in VARS_MES) < ANOS_MINIMOS:
            problemas.append(f"{nome}: alguma variável com menos de {ANOS_MINIMOS} anos válidos")
        if not g('txx_media') >= g('tmax_C_media') > g('tmed_C_media') > g('tmin_C_media') >= g('tnn_media'):
            problemas.append(f"{nome}: ordem TXx >= Tmax > Tmed > Tmin >= TNn não vale nas normais")
        if not g('rx1day_media') <= g('pr_mm_media') + 1e-6:
            problemas.append(f"{nome}: maior chuva diária normal acima do total normal")
        if not abs(g('dias_chuva_1mm_media') + g('dias_secos_media') - dm) < 0.1:
            problemas.append(f"{nome}: dias com chuva + dias secos não somam os dias do mês")
        if not g('graus_dia_media') >= 0:
            problemas.append(f"{nome}: graus-dia normal negativo")
        qs = [g(n_) for _, n_ in QUANTIS_CHUVA[:4]]
        if not all(np.isfinite(qs)) or any(b < a - 1e-6 for a, b in zip(qs, qs[1:])):
            problemas.append(f"{nome}: quintis da chuva fora de ordem ou ausentes")
        if not g('pr_mm_t33') <= g('pr_mm_t67'):
            problemas.append(f"{nome}: tercis da chuva fora de ordem")
        if g('soil_moisture_0_100cm_m3_m3_media') > UMIDADE_MAX_FISICA + 1e-4:
            problemas.append(f"{nome}: umidade 0-100 cm normal acima de {UMIDADE_MAX_FISICA}")
        if g('et_real_media') > g('et_potencial_media') + 1e-3:
            problemas.append(f"{nome}: ET real normal maior que ETo normal")
        for k in ESCALAS_SPEI:
            if g(f'spei{k}_n') >= ANOS_MINIMOS and not np.isfinite(g(f'spei{k}_alfa')):
                problemas.append(f"{nome}: SPEI-{k} sem ajuste apesar de {g(f'spei{k}_n'):.0f} anos válidos")
        q0 = g('spi1_q0')
        if np.isfinite(q0) and not 0 <= q0 <= 1:
            problemas.append(f"{nome}: q0 do SPI fora de 0 a 1")
        if np.isfinite(g('pr_p95_umido')) and not g('pr_p99_umido') >= g('pr_p95_umido'):
            problemas.append(f"{nome}: P99 da chuva menor que o P95")
        lim = {k: v for k, v in ee_l.get(nome, {}).items() if k != 'nome'}
        if 15 in DIAS_VERIFICACAO:
            v15 = [lim.get(f'{b}_d15') for b in ('tx90', 'tx10', 'tn90', 'tn10')]
            if None in v15 or not (v15[0] > v15[1] and v15[2] > v15[3] and v15[0] > v15[3]):
                problemas.append(f"{nome}: limiares do dia 15 fora de ordem (TX90 > TX10, TN90 > TN10)")

    tabela = []
    for q in pontos:
        e = ee_n.get(q['nome'], {})
        lim = ee_l.get(q['nome'], {})
        f = lambda k, nd=1: (round(float(e[k]), nd) if e.get(k) is not None else None)  # noqa: E731
        fl = lambda k: (round(float(lim[k]), 1) if lim.get(k) is not None else None)  # noqa: E731
        tabela.append({
            'ponto': q['nome'], 'chuva': f('pr_mm_media'), 'dp': f('pr_mm_dp'), 'q20': f('pr_mm_q20'),
            'q80': f('pr_mm_q80'), 'dias_chuva': f('dias_chuva_1mm_media'), 'rx1day': f('rx1day_media'),
            'P95_dia': f('pr_p95_umido'), 'Tmax': f('tmax_C_media'), 'Tmin': f('tmin_C_media'),
            'ETo': f('et_potencial_media'), 'GD': f('graus_dia_media', 0),
            'umid100': f('soil_moisture_0_100cm_m3_m3_media', 3), 'anos': f('pr_mm_n', 0),
            'spi3_alfa': f('spi3_alfa', 2), 'spei3_kappa': f('spei3_kappa', 3),
            'TX90_d15': fl('tx90_d15'), 'TN10_d15': fl('tn10_d15'),
        })
    print(f"\n   NORMAIS DO MÊS {mes:02d} NOS PONTOS ({p.replace('_', '-')}; chuva e ETo em mm/mês)")
    print(pd.DataFrame(tabela).to_string(index=False))

    # Assets já gravados entram na conferência quando existirem
    gravados_ok = None
    caminho = f"{asset_colecao('normais', p)}/{nome_imagem('normais', p, mes)}"
    if asset_existe(caminho):
        comp_g = comparar(amostrar(ee.Image(caminho), pontos), loc_n)
        gravados_ok = bool(comp_g.ok.all())
        print(f"   asset gravado {caminho.split('/')[-1]}: {int(comp_g.ok.sum())} de {len(comp_g)} valores iguais ao cálculo local")

    return {'numeros_ok': bool(comp.ok.all()), 'limiares_ok': bool(comp_l.ok.all()),
            'problemas': problemas, 'gravados_ok': gravados_ok}

def asset_existe(caminho):
    try:
        ee.data.getAsset(caminho)
        return True
    except ee.EEException:
        return False


def listar_nomes(pai):
    nomes, token = [], None
    while True:
        params = {'parent': pai, 'pageSize': 1000}
        if token:
            params['pageToken'] = token
        resp = ee.data.listAssets(params)
        for a in resp.get('assets', []):
            nomes.append((a.get('id') or a.get('name')).split('/')[-1])
        token = resp.get('nextPageToken')
        if not token:
            return nomes


def tarefas_ativas():
    return {t.get('description', ''): t.get('state') for t in ee.data.getTaskList()
            if t.get('state') in ('READY', 'RUNNING', 'SUBMITTED', 'PENDING')}


def garantir_asset(caminho, tipo):
    if asset_existe(caminho):
        return
    if SIMULAR:
        print(f"🧪 [SIMULAÇÃO] Criaria {tipo.lower()}: {caminho}")
        return
    ee.data.createAsset({'type': tipo}, caminho)
    print(f"📁 Criado: {caminho}")


def plano_de_exportacao():
    ativas = tarefas_ativas()
    plano = []
    for p in PERIODOS_A_GERAR:
        for tipo in ('normais', 'limiares'):
            if tipo == 'limiares' and not GERAR_LIMIARES:
                continue
            col = asset_colecao(tipo, p)
            existentes = set(listar_nomes(col)) if asset_existe(col) else set()
            for mes in MESES_A_GERAR:
                nome, desc = nome_imagem(tipo, p, mes), descricao_tarefa(tipo, p, mes)
                if desc in ativas:
                    situacao = 'na fila'
                elif nome in existentes and not SOBRESCREVER:
                    situacao = 'já existe'
                else:
                    situacao = 'enviar'
                plano.append({'periodo': p, 'tipo': tipo, 'mes': mes, 'colecao': col, 'nome': nome,
                              'descricao': desc, 'situacao': situacao})
    return pd.DataFrame(plano), ativas


def avisar_assets_antigos():
    for nome in ('Baseline_Historico_1961_1990', 'Baseline_Operacional_1991_2020'):
        caminho = f'{PASTA_DESTINO}/{nome}'
        if asset_existe(caminho):
            print(f"⚠️ {nome} existe e foi gerado pelo script anterior (SICSB e estatística sobre dias). "
                  f"O painel não o usa; apague-o pela aba Assets quando quiser.")


def acompanhar(descricoes):
    """Mostra o andamento e, no fim, o tempo e o consumo de EECU de cada tarefa."""
    if not descricoes:
        return {}
    print(f"\n⏳ Acompanhando {len(descricoes)} tarefa(s) a cada {INTERVALO_ACOMPANHAMENTO_S} s. "
          f"Pode interromper a célula: as tarefas continuam no Earth Engine.")
    alvo = set(descricoes)
    try:
        while True:
            info = {}
            for t in ee.data.getTaskList():
                d = t.get('description', '')
                if d in alvo and d not in info:         # a lista vem da mais nova para a mais antiga
                    info[d] = t
            estados = {d: t.get('state') for d, t in info.items()}
            cont = pd.Series(list(estados.values()), dtype='object').value_counts().to_dict()
            print(f"   {datetime.now(timezone(timedelta(hours=-3))).strftime('%H:%M')} {cont}")
            if info and not any(s in ('READY', 'RUNNING', 'SUBMITTED', 'PENDING') for s in estados.values()):
                falhas = [d for d, s in estados.items() if s != 'COMPLETED']
                consumo = {}
                for d, t in info.items():
                    eecu = t.get('batch_eecu_usage_seconds')
                    ini, fim = t.get('start_timestamp_ms'), t.get('update_timestamp_ms')
                    minutos = (fim - ini) / 60000 if ini and fim else float('nan')
                    consumo[d] = {'eecu_h': (eecu or 0) / 3600 if eecu is not None else float('nan'), 'min': minutos}
                    print(f"   {d}: {estados[d]} | {minutos:.0f} min | "
                          f"{consumo[d]['eecu_h']:.2f} EECU-hora(s)" + (f" | {t.get('error_message')}" if t.get('error_message') else ''))
                print("✅ Todas concluídas." if not falhas else f"❌ Não concluídas: {falhas}. Rode o script de novo para reenviar só essas.")
                return consumo
            time.sleep(INTERVALO_ACOMPANHAMENTO_S)
    except KeyboardInterrupt:
        print("⏸️ Acompanhamento interrompido; as tarefas seguem na fila.")
        return {}

def main():
    print(f"\n{'🧪 MODO SIMULAÇÃO: nada será criado ou exportado' if SIMULAR else '🚀 MODO REAL'}")
    print("Critérios para seguir: inventário sem duplicatas e com anos suficientes, números do Earth Engine "
          "iguais ao cálculo local, limiares dentro da faixa e nenhuma incoerência física.")

    resultados_inv = {p: inventario(p) for p in sorted(set(PERIODOS_A_GERAR) | {PERIODO_VERIFICACAO})}
    avisar_assets_antigos()
    plano, ativas = plano_de_exportacao()
    a_enviar = plano[plano.situacao == 'enviar']
    print(f"\n📦 PLANO: {len(plano)} imagens | enviar {len(a_enviar)} | já existem "
          f"{int((plano.situacao == 'já existe').sum())} | na fila {int((plano.situacao == 'na fila').sum())}")
    if len(a_enviar):
        print(a_enviar.groupby(['periodo', 'tipo']).mes.apply(lambda s: f"meses {s.min():02d} a {s.max():02d}").to_string())

    if SIMULAR:
        criterios = {f"inventário {p.replace('_', '-')} (sem datas duplicadas, só dias definitivos, "
                     f"pelo menos {ANOS_MINIMOS} anos válidos em todo mês)": r['ok']
                     for p, r in resultados_inv.items()}
        if VERIFICAR:
            for mes in MESES_VERIFICACAO:
                try:
                    r = verificar(PERIODO_VERIFICACAO, mes)
                except ee.EEException as e:
                    print(f"\n❌ Conferência do mês {mes:02d} não terminou: {str(e)[:200]}")
                    criterios[f"conferência numérica concluída (mês {mes:02d})"] = False
                    continue
                criterios[f"números EE = local (mês {mes:02d})"] = r['numeros_ok']
                criterios[f"limiares na faixa (mês {mes:02d})"] = r['limiares_ok']
                criterios[f"coerência física (mês {mes:02d})"] = not r['problemas']
                for pr in r['problemas']:
                    print(f"   ⚠️ {pr}")
                if r['gravados_ok'] is not None:
                    criterios[f"assets gravados = local (mês {mes:02d})"] = r['gravados_ok']
        print("\n=== CRITÉRIOS ===")
        for k, v in criterios.items():
            print(f"{'✅' if v else '❌'} {k}")
        if all(criterios.values()):
            print("\n✅ Tudo conferido. Para gerar de verdade, mude SIMULAR para False e rode de novo "
                  "depois que as tarefas do motor terminarem (com PILOTO = True vai só o mês 01 primeiro).")
        else:
            print("\n❌ Há critério não atendido. Não exporte antes de entender o motivo; mande esta saída.")
        return

    motor = [d for d in ativas if d.startswith('SAB_M_')]
    if motor and ESPERAR_MOTOR:
        print(f"\n⏸️ {len(motor)} tarefa(s) do motor ainda na fila (ex.: {motor[0]}). A fila do Earth Engine "
              f"é a mesma; enviar as normais agora atrasaria a previsão. Rode de novo quando terminarem "
              f"(ou mude ESPERAR_MOTOR para False).")
        return

    if PILOTO:
        a_enviar = a_enviar.groupby('tipo', sort=False).head(1)
        print(f"\n🧭 PILOTO: só {', '.join(a_enviar.descricao)}. Mede tempo e EECU antes de enviar o resto.")
    garantir_asset(PASTA_DESTINO, 'FOLDER')
    for col in sorted(set(a_enviar.colecao)):
        garantir_asset(col, 'IMAGE_COLLECTION')
    enviadas = []
    for _, lin in a_enviar.iterrows():
        img = (imagem_normais(lin.periodo, lin.mes) if lin.tipo == 'normais'
               else imagem_limiares(lin.periodo, lin.mes))
        extra = {'overwrite': True} if SOBRESCREVER else {}
        tarefa = ee.batch.Export.image.toAsset(
            image=img, description=lin.descricao, assetId=f'{lin.colecao}/{lin.nome}',
            region=limitesRegiao, crs=CRS, crsTransform=TRANSFORM, maxPixels=1e13, **extra)
        tarefa.start()
        enviadas.append(lin.descricao)
        print(f"🚀 {lin.descricao}")
    print(f"\n✔️ {len(enviadas)} tarefa(s) enviada(s). Acompanhe em https://code.earthengine.google.com/tasks")
    if ACOMPANHAR or PILOTO:
        consumo = acompanhar(enviadas)
        if PILOTO and consumo:
            todas = plano[plano.situacao == 'enviar']
            proj = 0.0
            for tipo, n in todas.groupby('tipo').size().items():
                medido = [v['eecu_h'] for d, v in consumo.items()
                          if d.startswith('SAB_NORMAIS_' if tipo == 'normais' else 'SAB_LIMIARES_') and v['eecu_h'] == v['eecu_h']]
                proj += (medido[0] if medido else float('nan')) * n
            print(f"\n📈 Projeção para as {len(todas)} imagens do plano: cerca de {proj:.0f} EECU-horas "
                  f"(cotas mensais não comerciais: Community 150, Contributor 1.000; veja a do projeto no Console do Google Cloud). "
                  f"Se couber, mude PILOTO para False e rode de novo; os meses prontos são pulados.")


main()
